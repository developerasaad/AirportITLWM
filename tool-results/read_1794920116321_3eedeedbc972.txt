    28→void AirportItlwmSkywalkInterface::deferWCLCountryUpdate()
    29→{
    30→    auto queue = instance->getWorkQueue();
    31→    if (wclStopping || !_fCommandGate || !queue)
    32→        return;
    33→    auto update = static_cast<CountryUpdate *>(IOMalloc(sizeof(CountryUpdate)));
    34→    if (!update)
    35→        return;
    36→    update->call = thread_call_allocate(wclCountryUpdateThread, update);
    37→    if (!update->call) {
    38→        IOFree(update, sizeof(*update));
    39→        return;
    40→    }
    41→    update->interface = this;
    42→    update->controller = instance;
    43→    update->gate = _fCommandGate;
    44→    update->queue = queue;
    45→    retain();
    46→    instance->retain();
    47→    update->gate->retain();
    48→    queue->retain();
    49→    thread_call_enter(update->call);
    50→}
    51→
    52→void AirportItlwmSkywalkInterface::wclCountryUpdateThread(void *arg, void *)
    53→{
    54→    auto update = static_cast<CountryUpdate *>(arg);
    55→    // Native country notifications synchronously query WCL. Holding the gate
    56→    // is necessary, but executing on the driver's workloop thread is forbidden.
    57→    update->gate->runAction(wclCountryUpdateGated, update);
    58→    thread_call_free(update->call);
    59→    update->interface->release();
    60→    update->controller->release();
    61→    update->gate->release();
    62→    update->queue->release();
    63→    IOFree(update, sizeof(*update));
    64→}

    66→IOReturn AirportItlwmSkywalkInterface::wclCountryUpdateGated(
    67→    OSObject *, void *arg, void *, void *, void *)
    68→{
    69→    auto update = static_cast<CountryUpdate *>(arg);
    70→    if (update->interface->wclStopping)
    71→        return kIOReturnNotReady;
    72→    if (!update->queue->inGate() || update->queue->onThread())
    73→        return kIOReturnNotPermitted;
    74→    update->interface->postMessage(APPLE80211_M_COUNTRY_CODE_CHANGED, nullptr, 0, false);
    75→    return kIOReturnSuccess;
    76→}

    86→void AirportItlwmSkywalkInterface::stopWCL()
    87→{
    88→    // Pending country tasks must observe shutdown under the same gate.
    89→    if (_fCommandGate)
    90→        _fCommandGate->runAction(stopWCLGated, this);
    91→    wclStopping = true;
    92→    if (joinTimer) {
    93→        joinTimer->cancelTimeout();
    94→        if (joinTimer->getWorkLoop())
    95→            joinTimer->getWorkLoop()->removeEventSource(joinTimer);
    96→        OSSafeReleaseNULL(joinTimer);
    97→    }
    98→    if (scanSource)
    99→        scanSource->cancelTimeout();
   100→    scanPending = joinPending = false;
   101→    joinDeferred = false;
   102→    explicit_bzero(&deferredJoin, sizeof(deferredJoin));
   103→    if (fHalService) {
   104→        ieee80211com *ic = fHalService->get80211Controller();
   105→        ic->ic_wcl_join_requested = false;
   106→        ic->ic_wcl_scan_requested = false;
   107→        explicit_bzero(ic->ic_psk, sizeof(ic->ic_psk));
   108→    }
   109→}

   128→IOReturn AirportItlwmSkywalkInterface::updateWCLLinkQualityGated(
   129→    OSObject *, void *arg0, void *arg1, void *, void *)
   130→{
   131→    auto self = static_cast<AirportItlwmSkywalkInterface *>(arg0);
   132→    if (self->wclStopping || !self->fHalService)
   133→        return kIOReturnNotReady;
   134→    auto queue = self->instance->getWorkQueue();
   135→    if (!queue || !queue->inGate())
   136→        return kIOReturnNotPermitted;
   137→    auto interface = self->mExpansionData ? self->getBSDInterface() : nullptr;
   138→    if (!interface) {
   139→        self->wclLinkQualityInterface = nullptr;
   140→        self->wclReportedLinkQuality = -2;
   141→        return kIOReturnNotReady;
   142→    }
   143→    auto ic = self->fHalService->get80211Controller();
   144→    int quality = -2;
   145→    if (self->portAuthorized && self->linkState() == kIO80211NetworkLinkUp &&
   146→        (ifnet_flags(interface) & (IFF_UP | IFF_RUNNING)) == (IFF_UP | IFF_RUNNING) &&
   147→        ic && ic->ic_state == IEEE80211_S_RUN && ic->ic_bss &&
   148→        IEEE80211_ADDR_EQ(ic->ic_bss->ni_bssid, self->joinBSSID)) {
   149→        int rssi = IWM_MIN_DBM + ic->ic_bss->ni_rssi;
   150→        quality = rssi >= -70 ? 100 : (rssi >= -85 ? 50 : 20);
   151→    }
   152→    if (arg1 || quality != self->wclReportedLinkQuality ||
   153→        interface != self->wclLinkQualityInterface) {
   154→        self->setLQM(static_cast<unsigned long long>(static_cast<long long>(quality)));
   155→        self->wclReportedLinkQuality = quality;
   156→        self->wclLinkQualityInterface = interface;
   157→    }
   158→    return kIOReturnSuccess;
   159→}

   168→IOReturn AirportItlwmSkywalkInterface::wclCommand(OSObject *, void *a0, void *a1, void *a2, void *)
   169→{
   170→    auto self = static_cast<AirportItlwmSkywalkInterface *>(a0);
   171→    if (self->wclStopping)
   172→        return kIOReturnNotReady;
   173→    switch (uintptr_t(a1)) {
   174→        case 13: return self->copyWCLChannels(a2);
   175→        case 14: return self->copyWCLTrafficCounters(a2);
   176→        case 15:
   177→            return self->fHalService->setScanHomeAwayTime(
   178→                AirportWCL::read32(static_cast<const uint8_t *>(a2)));
   179→        case 12:
   180→            self->handleWCLEvent(int(uintptr_t(a2)), nullptr);
   181→            return kIOReturnSuccess;
   182→        case 11: self->clearWCLPMKSA(); return kIOReturnSuccess;
   183→        case 8: return self->installWCLKey(static_cast<apple80211_key *>(a2));
   184→        case 9: return self->accessWCLRSN(static_cast<apple80211_rsn_ie_data *>(a2), true);
   185→        case 10: return self->accessWCLRSN(static_cast<apple80211_rsn_ie_data *>(a2), false);
   186→        case 0: return self->beginWCLScan(static_cast<apple80211ScanRequest *>(a2));
   187→        case 1: return self->beginWCLJoin(static_cast<apple80211AssocCandidates *>(a2));
   188→        case 2: return self->abortWCLJoin();
   189→        case 3: return self->leaveWCLNetwork();
   190→        case 6: return self->copyWCLExtendedBss(a2);
   191→        case 7:
   192→            if (!a2 || !self->wclBssValid || !self->wclAssociatedBeacons)
   193→                return kIOReturnNotReady;
   194→            memcpy(a2, self->wclBeaconInfo, sizeof(self->wclBeaconInfo));
   195→            return kIOReturnSuccess;
   196→        case 5:
   197→            if (!a2 || !self->wclBssValid)
   198→                return kIOReturnNotReady;
   199→            memcpy(a2, self->wclBssInfo, sizeof(self->wclBssInfo));
   200→            return kIOReturnSuccess;
   201→        case 4:
   202→            self->finishWCLScan(kIOReturnAborted);
   203→            return kIOReturnSuccess;
   204→    }
   205→    return kIOReturnUnsupported;
   206→}

   237→IOReturn AirportItlwmSkywalkInterface::beginWCLScan(apple80211ScanRequest *request)
   238→{
   239→    AirportWCL::ScanRequest scan;
   240→    if (!AirportWCL::decodeScan(request, 0x1550, scan))
   241→        return kIOReturnBadArgument;
   242→    ieee80211com *ic = fHalService->get80211Controller();
   243→    if (!(ic->ic_if.if_flags & IFF_RUNNING))
   244→        return kIOReturnNotReady;
   245→    if (scanPending || joinPending || (ic->ic_flags & IEEE80211_F_BGSCAN))
   246→        return kIOReturnBusy;
   247→    if (ic->ic_state == IEEE80211_S_AUTH || ic->ic_state == IEEE80211_S_ASSOC)
   248→        return kIOReturnBusy;
   249→    if (scan.channelCount) {
   250→        bool usable = false;
   251→        for (unsigned channel = 1; channel <= IEEE80211_CHAN_MAX; ++channel)
   252→            if (ic->ic_channels[channel].ic_flags &&
   253→                AirportWCL::scanChannelSelected(scan.channels, true, channel))
   254→                usable = true;
   255→        if (!usable)
   256→            return kIOReturnUnsupported;
   257→    }
   258→    scanPending = true;
   259→    scanCompleted = false;
   260→    scanSkipCompletion = ic->ic_state == IEEE80211_S_SCAN &&
   261→        ((scan.type != 2 && scan.ssidLength != 0) ||
   262→         ic->ic_wcl_scan_ssid_length != 0 || ic->ic_des_esslen != 0);
   263→    ic->ic_wcl_scan_requested = true;
   264→    ic->ic_wcl_scan_active = scan.type != 2;
   265→    ic->ic_wcl_scan_ssid_length = scan.type == 2 ? 0 : scan.ssidLength;
   266→    memcpy(ic->ic_wcl_scan_ssid, scan.ssid, sizeof(scan.ssid));
   267→    ic->ic_wcl_scan_restricted = scan.channelCount != 0;
   268→    memcpy(ic->ic_wcl_scan_channels, scan.channels, sizeof(scan.channels));
   269→    scanSource->setTimeoutMS(20000);
   270→    if (ic->ic_state == IEEE80211_S_RUN) {
   271→        ieee80211_begin_cache_bgscan(&ic->ic_if);
   272→        if (!(ic->ic_flags & IEEE80211_F_BGSCAN)) {
   273→            scanPending = false;
   274→            ic->ic_wcl_scan_requested = false;
   275→            scanSource->cancelTimeout();
   276→            return kIOReturnBusy;
   277→        }
   278→    } else if (ic->ic_state != IEEE80211_S_SCAN) {
   279→        int error = ic->ic_newstate(ic, IEEE80211_S_SCAN, -1);
   280→        if (error) {
   281→            scanPending = false;
   282→            ic->ic_wcl_scan_requested = false;
   283→            scanSource->cancelTimeout();
   284→            return kIOReturnError;
   285→        }
   286→    }
   287→    return kIOReturnSuccess;
   288→}

   290→void AirportItlwmSkywalkInterface::finishWCLScan(IOReturn status)
   291→{
   292→    if (!scanPending)
   293→        return;
   294→    scanPending = false;
   295→    scanCompleted = false;
   296→    scanSource->cancelTimeout();
   297→    fHalService->get80211Controller()->ic_wcl_scan_requested = false;
   298→    instance->postMessage(this, AirportWCL::ScanDone, &status, sizeof(status), true);
   299→}

   301→void AirportItlwmSkywalkInterface::completeWCLScan()
   302→{
   303→    if (!wclStopping)
   304→        finishWCLScan(scanCompleted ? kIOReturnSuccess : kIOReturnTimeout);
   305→}

   307→void AirportItlwmSkywalkInterface::publishWCLBeacon(const ieee80211_node *node, const uint8_t *ies, size_t length)
   308→{
   309→    if (!node || node->ni_esslen > 32 || !AirportWCL::validIEs(ies, length))
   310→        return;
   311→    ieee80211com *ic = fHalService->get80211Controller();
   312→    if (!node->ni_chan || node->ni_chan == IEEE80211_CHAN_ANYC)
   313→        return;
   314→    unsigned channel = ieee80211_chan2ieee(ic, node->ni_chan);
   315→    if (!channel || channel > 196)
   316→        return;
   317→    size_t size = sizeof(AirportWCL::BeaconMetadata) + length;
   318→    auto metadata = static_cast<AirportWCL::BeaconMetadata *>(IOMalloc(size));
   319→    if (!metadata)
   320→        return;
   321→    *metadata = {};
   322→    metadata->ieLength = uint32_t(length);
   323→    metadata->channelSpec = uint16_t(channel | (IEEE80211_IS_CHAN_5GHZ(node->ni_chan) ? 0xd000 : 0x1000));
   324→    metadata->channel = channel;
   325→    metadata->ssidLength = node->ni_esslen;
   326→    memcpy(metadata->ssid, node->ni_essid, node->ni_esslen);
   327→    memcpy(metadata->bssid, node->ni_bssid, 6);
   328→    metadata->rssi = IWM_MIN_DBM + node->ni_rssi;
   329→    metadata->interval = node->ni_intval;
   330→    metadata->capability = node->ni_capinfo;
   331→    metadata->flags = AirportWCL::SSIDPresent | AirportWCL::RSSIValid;
   332→    memcpy(metadata + 1, ies, length);
   333→    metadata->ieLength = uint32_t(AirportWCL::selectWPA2TransitionMode(
   334→        reinterpret_cast<uint8_t *>(metadata + 1), length));
   335→    size_t messageSize = sizeof(*metadata) + metadata->ieLength;
   336→    if (assocNotified && IEEE80211_ADDR_EQ(node->ni_bssid, joinBSSID)) {
   337→        wclBssValid = messageSize <= sizeof(wclBssInfo);
   338→        if (wclBssValid) {
   339→            bzero(wclBssInfo, sizeof(wclBssInfo));
   340→            memcpy(wclBssInfo, metadata, messageSize);
   341→        }
   342→    }
   343→    instance->postMessage(this, AirportWCL::ScanResult, metadata, messageSize, true);
   344→    IOFree(metadata, size);
   345→}

   347→void AirportItlwmSkywalkInterface::clearWCLPMKSA()
   348→{
   349→    auto ic = fHalService->get80211Controller();
   350→    ieee80211_pmk *pmk;
   351→    while ((pmk = TAILQ_FIRST(&ic->ic_pmksa))) {
   352→        TAILQ_REMOVE(&ic->ic_pmksa, pmk, pmk_next);
   353→        explicit_bzero(pmk, sizeof(*pmk));
   354→        ::free(pmk);
   355→    }
   356→}

   363→IOReturn AirportItlwmSkywalkInterface::installWCLKey(apple80211_key *key)
   364→{
   365→    static_assert(sizeof(apple80211_key) == 0x94, "WCL cipher key layout");
   366→    if (!key || !AirportWCL::validCipherKey(key->key_cipher_type, key->key_len,
   367→            key->key_flags, key->key_index, key->key_rsc_len))
   368→        return kIOReturnBadArgument;
   369→    auto ic = fHalService->get80211Controller();
   370→    auto ni = ic->ic_bss;
   371→    if (!ic->ic_wcl_enterprise || !ni || ic->ic_state != IEEE80211_S_RUN)
   372→        return kIOReturnNotReady;
   373→    if (key->key_cipher_type == APPLE80211_CIPHER_PMK ||
   374→        key->key_cipher_type == APPLE80211_CIPHER_PMKSA ||
   375→        key->key_cipher_type == APPLE80211_CIPHER_MSK) {
   376→        // WCL's native user-client rejects PTK/GTK, and delivers a PMK/MSK
   377→        // instead. Use the existing net80211 four-way/group-key handshake.
   378→        if (!ieee80211_is_8021x_akm((enum ieee80211_akm)ni->ni_rsnakms))
   379→            return kIOReturnBadArgument;
   380→        uint8_t any = 0;
   381→        for (auto byte : key->key_ea.octet) any |= byte;
   382→        if (any && !IEEE80211_ADDR_EQ(key->key_ea.octet, ni->ni_bssid))
   383→            return kIOReturnBadArgument;
   384→        auto pmk = ieee80211_pmksa_add(ic, (enum ieee80211_akm)ni->ni_rsnakms,
   385→                                     ni->ni_macaddr, key->key, 0);
   386→        IOReturn result = pmk ? kIOReturnSuccess : kIOReturnNoMemory;
   387→        return result;
   388→    }
   389→    return kIOReturnUnsupported;
   390→}

   392→IOReturn AirportItlwmSkywalkInterface::accessWCLRSN(apple80211_rsn_ie_data *data, bool write)
   393→{
   394→    if (!data) return kIOReturnBadArgument;
   395→    auto ic = fHalService->get80211Controller();
   396→    if (write) {
   397→        if (!AirportWCL::validRSNOverride(data->ie, data->len))
   398→            return kIOReturnBadArgument;
   399→        // net80211 constructs negotiated IEs for its own key handshake.
   400→        return kIOReturnUnsupported;
   401→
   402→    }
   403→    const uint8_t *ie = ic->ic_bss ? ic->ic_bss->ni_rsnie : nullptr;
   404→    if (!ie) return kIOReturnNotReady;
   405→    size_t length = size_t(ie[1]) + 2;
   406→    if (length > sizeof(data->ie)) return kIOReturnBadArgument;
   407→    bzero(data, sizeof(*data));
   408→    data->version = APPLE80211_VERSION;
   409→    data->len = length;
   410→    memcpy(data->ie, ie, length);
   411→    return kIOReturnSuccess;
   412→}

   423→IOReturn AirportItlwmSkywalkInterface::startWCLJoin(AirportWCL::JoinRequest &join)
   424→{
   425→    auto auth = AirportWCL::classifyLegacyAuth(join);
   426→    bool personal = auth == AirportWCL::LegacyAuth::Personal;
   427→    bool enterprise = auth == AirportWCL::LegacyAuth::Enterprise;
   428→    bool supported = auth != AirportWCL::LegacyAuth::Invalid;
   429→    ieee80211com *ic = fHalService->get80211Controller();
   430→    // Applying a private MAC restarts firmware asynchronously. WCL can issue
   431→    // the association before that restart has completed; keep one request
   432→    // until the HAL reports READY rather than making the user retry it.
   433→    if (supported && !joinPending && ic->ic_wcl_mac_reconfig &&
   434→        (ic->ic_if.if_flags & IFF_UP)) {
   435→        deferredJoin = join;
   436→        memcpy(joinBSSID, join.bssid, sizeof(joinBSSID));
   437→        explicit_bzero(&join, sizeof(join));
   438→        joinDeferred = joinPending = true;
   439→        joinTimer->setTimeoutMS(30000);
   440→        return kIOReturnSuccess;
   441→    }
   442→    if (!supported || !(ic->ic_if.if_flags & IFF_RUNNING) || joinPending || scanPending) {
   443→        explicit_bzero(&join, sizeof(join));
   444→        return !supported ? kIOReturnUnsupported : kIOReturnBusy;
   445→    }
   446→    if (ic->ic_state == IEEE80211_S_RUN || ic->ic_state == IEEE80211_S_AUTH || ic->ic_state == IEEE80211_S_ASSOC)
   447→        leaveWCLNetwork();
   448→    ieee80211_disable_rsn(ic);
   449→    ieee80211_disable_wep(ic);
   450→    explicit_bzero(ic->ic_psk, sizeof(ic->ic_psk));
   451→    clearWCLPMKSA();
   452→    ic->ic_flags &= ~(IEEE80211_F_AUTO_JOIN | IEEE80211_F_PSK);
   453→    ic->ic_wcl_enterprise = enterprise;
   454→
   455→    if (personal || enterprise) {
   456→        ieee80211_wpaparams wpa = {};
   457→        wpa.i_enabled = 1;
   458→        if (join.upperAuth & (APPLE80211_AUTHTYPE_WPA | APPLE80211_AUTHTYPE_WPA_PSK))
   459→            wpa.i_protos |= IEEE80211_WPA_PROTO_WPA1;
   460→        if (join.upperAuth & (APPLE80211_AUTHTYPE_WPA2 | APPLE80211_AUTHTYPE_WPA2_PSK |
   461→                              APPLE80211_AUTHTYPE_SHA256_PSK | APPLE80211_AUTHTYPE_SHA256_8021X))
   462→            wpa.i_protos |= IEEE80211_WPA_PROTO_WPA2;
   463→        if (join.upperAuth & (APPLE80211_AUTHTYPE_WPA_PSK | APPLE80211_AUTHTYPE_WPA2_PSK))
   464→            wpa.i_akms |= IEEE80211_WPA_AKM_PSK;
   465→        if (join.upperAuth & (APPLE80211_AUTHTYPE_WPA | APPLE80211_AUTHTYPE_WPA2))
   466→            wpa.i_akms |= IEEE80211_WPA_AKM_8021X;
   467→        if (join.upperAuth & APPLE80211_AUTHTYPE_SHA256_PSK) {
   468→            wpa.i_akms |= IEEE80211_WPA_AKM_SHA256_PSK;
   469→            wpa.i_akms |= IEEE80211_WPA_AKM_PSK;
   470→        }
   471→        if (join.upperAuth & APPLE80211_AUTHTYPE_SHA256_8021X)
   472→            wpa.i_akms |= IEEE80211_WPA_AKM_SHA256_8021X;
   473→        wpa.i_ciphers = IEEE80211_WPA_CIPHER_CCMP | IEEE80211_WPA_CIPHER_TKIP;
   474→        if (join.rsnLength >= 8 && join.rsn[0] == IEEE80211_ELEMID_RSN &&
   475→            join.rsn[4] == 0 && join.rsn[5] == 0x0f && join.rsn[6] == 0xac) {
   476→            if (join.rsn[7] == 2) wpa.i_groupcipher = IEEE80211_WPA_CIPHER_TKIP;
   477→            if (join.rsn[7] == 4) wpa.i_groupcipher = IEEE80211_WPA_CIPHER_CCMP;
   478→        }
   479→        int error = ieee80211_ioctl_setwpaparms(ic, &wpa);
   480→        if (error && error != ENETRESET) {
   481→            explicit_bzero(&join, sizeof(join));
   482→            return kIOReturnBadArgument;
   483→        }
   484→        if (personal) {
   485→            memcpy(ic->ic_psk, join.key, sizeof(ic->ic_psk));
   486→            ic->ic_flags |= IEEE80211_F_PSK;
   487→        } else if (join.keyLength == IEEE80211_PMK_LEN) {
   488→            bool cached = true;
   489→            if (join.upperAuth & (APPLE80211_AUTHTYPE_WPA | APPLE80211_AUTHTYPE_WPA2))
   490→                cached = ieee80211_pmksa_add(ic, IEEE80211_AKM_8021X, join.bssid, join.key, 0) != nullptr;
   491→            if (cached && (join.upperAuth & APPLE80211_AUTHTYPE_SHA256_8021X))
   492→                cached = ieee80211_pmksa_add(ic, IEEE80211_AKM_SHA256_8021X, join.bssid, join.key, 0) != nullptr;
   493→            if (!cached) {
   494→                clearWCLPMKSA();
   495→                explicit_bzero(&join, sizeof(join));
   496→                return kIOReturnNoMemory;
   497→            }
   498→        }
   499→    } else if (auth == AirportWCL::LegacyAuth::Wep) {
   500→        ieee80211_nwkey keys = {};
   501→        keys.i_wepon = IEEE80211_NWKEY_WEP;
   502→        keys.i_defkid = join.keyIndex + 1;
   503→        keys.i_key[join.keyIndex].i_keylen = join.keyLength;
   504→        keys.i_key[join.keyIndex].i_keydat = join.key;
   505→        int error = ieee80211_ioctl_setnwkeys(ic, &keys);
   506→        if (error && error != ENETRESET) {
   507→            explicit_bzero(&join, sizeof(join));
   508→            return kIOReturnBadArgument;
   509→        }
   510→    }
   511→    ic->ic_des_esslen = join.ssidLength;
   512→    memcpy(ic->ic_des_essid, join.ssid, sizeof(ic->ic_des_essid));
   513→    memcpy(ic->ic_des_bssid, join.bssid, 6);
   514→    memcpy(joinBSSID, join.bssid, 6);
   515→    ic->ic_flags |= IEEE80211_F_DESBSSID;
   516→    ic->ic_wcl_join_requested = true;
   517→    ic->ic_wcl_scan_requested = false;
   518→    current_authtype_lower = join.lowerAuth;
   519→    current_authtype_upper = join.upperAuth;
   520→    explicit_bzero(&join, sizeof(join));
   521→    wclBssValid = false;
   522→    joinPending = true;
   523→    assocNotified = firstBeaconNotified = portAuthorized = false;
   524→    wclAssociatedBeacons = 0;
   525→    bzero(wclBeaconInfo, sizeof(wclBeaconInfo));
   526→    disassocIsVoluntary = false;
   527→    joinTimer->setTimeoutMS(30000);
   528→    if (ic->ic_state != IEEE80211_S_SCAN && ic->ic_newstate(ic, IEEE80211_S_SCAN, -1)) {
   529→        joinPending = false;
   530→        joinTimer->cancelTimeout();
   531→        ic->ic_wcl_join_requested = false;
   532→        clearWCLPMKSA();
   533→        ic->ic_wcl_enterprise = false;
   534→        explicit_bzero(ic->ic_psk, sizeof(ic->ic_psk));
   535→        return kIOReturnError;
   536→    }
   537→    return kIOReturnSuccess;
   538→}

   540→void AirportItlwmSkywalkInterface::resumeWCLJoin()
   541→{
   542→    if (!joinDeferred || wclStopping)
   543→        return;
   544→    AirportWCL::JoinRequest join = deferredJoin;
   545→    explicit_bzero(&deferredJoin, sizeof(deferredJoin));
   546→    joinDeferred = joinPending = false;
   547→    joinTimer->cancelTimeout();
   548→    IOReturn status = startWCLJoin(join);
   549→    explicit_bzero(&join, sizeof(join));
   550→    if (status != kIOReturnSuccess) {
   551→        joinPending = true;
   552→        finishWCLJoin(1, 0, status);
   553→    }

   624→void AirportItlwmSkywalkInterface::handleWCLLink(bool up)
   625→{
   626→    if (wclStopping)
   627→        return;
   628→    portAuthorized = up;
   629→    if (!up && wclLinkIndicated) {
   630→        AirportWCL::LinkStatusEvent event = {};
   631→        memcpy(event.bssid, joinBSSID, sizeof(event.bssid));

   640→}
   641→
   642→void AirportItlwmSkywalkInterface::handleWCLEvent(int code, void *data)
   643→{
   644→    if (wclStopping)
   645→        return;
   646→    // Firmware initialization runs on systq, outside the native PostOffice gate.
   647→    // Serialize only lifecycle notifications; never run synchronous WCL ioctls here.
   648→    if ((code == IEEE80211_EVT_DRIVER_RESET_BEGIN || code == IEEE80211_EVT_DRIVER_READY ||
   649→         code == IEEE80211_EVT_DRIVER_STOPPED) && !instance->getWorkQueue()->inGate()) {
   650→        _fCommandGate->runAction(wclCommand, this, reinterpret_cast<void *>(12),
   651→                                reinterpret_cast<void *>(uintptr_t(code)));
   652→        return;
   653→    }
   654→    ieee80211com *ic = fHalService->get80211Controller();
   655→    switch (code) {
   656→        case IEEE80211_EVT_COUNTRY_CODE_UPDATE:
   657→            deferWCLCountryUpdate();
   658→            break;
   659→        case IEEE80211_EVT_DRIVER_RESET_BEGIN: {
   660→            portAuthorized = false;
   661→            updateWCLLinkQuality(true);
   662→            if (ic->ic_wcl_mac_reconfig) {
   663→                break;
   664→            }
   665→            uint32_t reset = 0; // Native resetInterfacesBegin.
   666→            instance->postMessage(this, AirportWCL::ResetInterface, &reset, sizeof(reset), true);
   667→            break;
   668→        }
   669→        case IEEE80211_EVT_DRIVER_READY:
   670→        case IEEE80211_EVT_DRIVER_STOPPED: {
   671→            bool ready = code == IEEE80211_EVT_DRIVER_READY;
   672→            if (!ready) {
   673→                finishWCLScan(ic->ic_wcl_mac_reconfig ? kIOReturnAborted : kIOReturnOffline);
   674→                finishWCLJoin(1, 0, kIOReturnOffline);
   675→                portAuthorized = false;
   676→                updateWCLLinkQuality(true);
   677→                ic->ic_wcl_join_requested = false;
   678→                clearWCLPMKSA();
   679→                ic->ic_wcl_enterprise = false;
   680→                explicit_bzero(ic->ic_psk, sizeof(ic->ic_psk));
   681→            }
   682→            // An address application is requested by WCL itself, not a new
   683→            // adapter reset. Replaying firmware-load events here rotates the
   684→            // scan MAC again and recursively restarts the card.
   685→            if (AirportWCL::consumeMacReconfiguration(ic->ic_wcl_mac_reconfig, ready)) {
   686→                if (ready)
   687→                    resumeWCLJoin();
   688→                break;
   689→            }
   690→            if (ready) {
   691→                uint32_t reset = 1; // Native resetInterfacesComplete, only after HAL success.
   692→                instance->postMessage(this, AirportWCL::ResetInterface, &reset, sizeof(reset), true);
   693→            }
   694→            AirportWCL::DriverAvailability event = {};
   695→            event.version = 3;
   696→            event.available = ready;
   697→            event.reason = ready ? 0xe0821803 : 0xe0821804;
   698→            instance->postMessage(this, AirportWCL::DriverAvailable, &event, sizeof(event), true);
   699→            if (ready)
   700→                resumeWCLJoin();
   701→            break;
   702→        }
   703→        case IEEE80211_EVT_BEACON: {
   704→            auto beacon = static_cast<ieee80211_beacon_event *>(data);
   705→            if (!beacon)
   706→                break;
   707→            // Retain the actual associated beacon, including its advancing TSF.
   708→            // WCL polls BEACON_INFO to distinguish fresh reception from a stale BSS.
   709→            if (assocNotified && (joinPending || portAuthorized) &&
   710→                IEEE80211_ADDR_EQ(beacon->node->ni_bssid, joinBSSID) &&
   711→                beacon->frame && beacon->frame_length >= 36 &&
   712→                beacon->frame_length <= 0x800 &&
   713→                (beacon->frame[0] & 0xfc) == IEEE80211_FC0_SUBTYPE_BEACON) {
   714→                uint32_t length = uint32_t(beacon->frame_length);
   715→                bzero(wclBeaconInfo, sizeof(wclBeaconInfo));
   716→                memcpy(wclBeaconInfo, &length, sizeof(length));
   717→                memcpy(wclBeaconInfo + 4, beacon->frame, length);
   718→                ++wclAssociatedBeacons;
   719→                ++wclLqmBeacons;
   720→                // WCL has no legacy LQMData instance at createLQMData().
   721→                // Send the snapshot consumed by WCL instead of legacy 227/229.
   722→                if (portAuthorized && (wclLqmBeacons % 8 == 0)) {
   723→#if __IO80211_TARGET == __MAC_15_2
   724→                    apple80211_rssi_data rssi = {};
   725→                    if (getRSSI(&rssi) == kIOReturnSuccess) {
   726→                        AirportWCL::SequoiaLqmUpdate update = {};
   727→                        update.rssiValid = 1;
   728→                        update.rssi = rssi.aggregate_rssi;
   729→                        instance->postMessage(this, AirportWCL::LqmUpdate,
   730→                            &update, sizeof(update), true);
   731→                    }
   732→#else
   733→                    AirportWCL::LqmBeaconUpdate update;
   734→                    if (AirportWCL::makeBeaconUpdate(
   735→                            wclLqmBeacons - wclLqmReportedBeacons, update)) {
   736→                        instance->postMessage(this, AirportWCL::LqmUpdate,
   737→                            &update, sizeof(update), true);
   738→                        wclLqmReportedBeacons = wclLqmBeacons;
   739→                    }
   740→#endif
   741→                    updateWCLLinkQuality();
   742→                }
   743→            }
   744→            if (scanPending || joinPending || ic->ic_state == IEEE80211_S_SCAN ||
   745→                (portAuthorized && IEEE80211_ADDR_EQ(beacon->node->ni_bssid, joinBSSID)))
   746→                publishWCLBeacon(beacon->node, beacon->ies, beacon->length);
   747→            if (joinPending && assocNotified && !firstBeaconNotified &&
   748→                IEEE80211_ADDR_EQ(beacon->node->ni_bssid, joinBSSID)) {
   749→                AirportWCL::FirstBeaconEvent event = {};
   750→                memcpy(event.peer.bssid, joinBSSID, 6);
   751→                if (!wclBssValid)
   752→                    break;
   753→                firstBeaconNotified = true;
   754→                AirportWCL::LinkStatusEvent link = {};
   755→                memcpy(link.bssid, joinBSSID, sizeof(link.bssid));
   756→                link.up = 1;
   757→                link.reason = 255;
   758→                wclLinkIndicated = true;
   759→                instance->postMessage(this, AirportWCL::LinkStatus, &link, sizeof(link), true);
   760→                instance->postMessage(this, AirportWCL::FirstBeacon, &event, sizeof(event), true);
   761→                if (portAuthorized)
   762→                    finishWCLJoin(0, 0, kIOReturnSuccess);
   763→            }
   764→            break;
   765→        }
   766→        case IEEE80211_EVT_SCAN_DONE:
   767→            if (scanPending) {
   768→                if (scanSkipCompletion)
   769→                    scanSkipCompletion = false;
   770→                else {
   771→                    scanCompleted = true;
   772→                    scanSource->setTimeoutMS(0);
   773→                }
   774→            }
   775→            break;
   776→        case IEEE80211_EVT_STA_ASSOC_DONE:
   777→            if (joinPending && !assocNotified) {
   778→                AirportWCL::AssociationEvent event = {};
   779→                event.status = ic->ic_assoc_status;
   780→                memcpy(event.bssid, joinBSSID, 6);
   781→                event.driverStatus = event.status ? kIOReturnError : kIOReturnSuccess;
   782→                assocNotified = !event.status;
   783→                instance->postMessage(this, AirportWCL::AssocComplete, &event, sizeof(event), true);
   784→                if (event.status)
   785→                    finishWCLJoin(event.status, 0, kIOReturnError);
   786→            }
   787→            break;
   788→        case IEEE80211_EVT_STA_DEAUTH: {
   789→            portAuthorized = false;
   790→            updateWCLLinkQuality(true);
   791→            AirportWCL::DeauthEvent event = {};
   792→            event.reason = ic->ic_deauth_reason;
   793→            if (ic->ic_bss)
   794→                memcpy(event.bssid, ic->ic_bss->ni_bssid, 6);
   795→            instance->postMessage(this, AirportWCL::Deauth, &event, sizeof(event), true);
   796→            finishWCLJoin(1, event.reason, kIOReturnOffline);
   797→            break;
   798→        }
   799→        default: break;
   800→    }
   801→}

   854→IOReturn AirportItlwmSkywalkInterface::copyWCLExtendedBss(void *data)
   855→{
   856→    ieee80211com *ic = fHalService->get80211Controller();
   857→    if (!data || !ic->ic_bss || ic->ic_state != IEEE80211_S_RUN)
   858→        return kIOReturnNotReady;
   859→    static_assert(sizeof(apple80211_rate_set_data) == 0xbc, "Extended BSS rates size");
   860→    static_assert(sizeof(apple80211_mcs_index_set_data) == 0x10, "Extended BSS MCS size");
   861→    auto bytes = static_cast<uint8_t *>(data);
   862→#if __IO80211_TARGET == __MAC_15_2
   863→    bzero(bytes, 0x1e0);
   864→#else
   865→    bzero(bytes, 0x214);
   866→#endif
   867→    IOReturn status = getRATE_SET(reinterpret_cast<apple80211_rate_set_data *>(bytes));
   868→    if (status)
   869→        return status;
   870→    status = getMCS_INDEX_SET(reinterpret_cast<apple80211_mcs_index_set_data *>(bytes + 0xbc));
   871→    if (status)
   872→        return status;
   873→    apple80211_vht_mcs_index_set_data vht = {};
   874→    vht.version = APPLE80211_VERSION;
   875→    vht.mcs_map = 0xffff;
   876→    if (ic->ic_bss->ni_flags & IEEE80211_NODE_VHT)
   877→        vht.mcs_map = ic->ic_bss->ni_vht_mcsinfo.tx_mcs_map;
   878→    memcpy(bytes + 0xcc, &vht, sizeof(vht));
   879→    // Native updateMCSSet uses version + a 16-bit HE MCS/NSS map (8 bytes).
   880→    uint32_t heVersion = APPLE80211_VERSION;
   881→    uint16_t heMap = 0xffff;
   882→    if (ic->ic_bss->ni_flags & IEEE80211_NODE_HE) {
   883→        bool wide = ic->ic_bss->ni_chw == IEEE80211_CHAN_WIDTH_160;
   884→        heMap = htole16(ieee80211_he_mcs_intersection(
   885→            le16toh(wide ? ic->ic_he_mcs_nss_supp.rx_mcs_160 : ic->ic_he_mcs_nss_supp.rx_mcs_80),
   886→            le16toh(wide ? ic->ic_bss->ni_he_mcs_nss_supp.tx_mcs_160 : ic->ic_bss->ni_he_mcs_nss_supp.tx_mcs_80)));
   887→    }
   888→    memcpy(bytes + 0xd4, &heVersion, sizeof(heVersion));
   889→    memcpy(bytes + 0xd8, &heMap, sizeof(heMap));
   890→    if (ic->ic_flags & IEEE80211_F_RSNON)
   891→#if __IO80211_TARGET == __MAC_15_2
   892→        ieee80211_add_rsn(bytes + 0xdd, ic, ic->ic_bss);
   893→#else
   894→        ieee80211_add_rsn(bytes + 0x113, ic, ic->ic_bss);
   895→#endif
   896→    return kIOReturnSuccess;
   897→}

   922→IOReturn AirportItlwmSkywalkInterface::getWCL_CHANNELS_INFO(apple80211ChannelInfo *data)
   923→{
   924→    if (!data)
   925→        return kIOReturnBadArgument;
   926→    auto info = reinterpret_cast<AirportWCL::ChannelInfo *>(data);
   927→    bzero(info, sizeof(*info));
   928→    ieee80211com *ic = fHalService->get80211Controller();
   929→    for (unsigned i = 1; i < IEEE80211_CHAN_MAX && info->count < 400; ++i) {
   930→        if (!isset(ic->ic_chan_active, i))
   931→            continue;
   932→        const ieee80211_channel *channel = &ic->ic_channels[i];
   933→        auto &entry = info->channels[info->count++];

   967→            channel->ic_flags & IEEE80211_CHAN_PASSIVE,
   968→            IEEE80211_IS_CHAN_HT40(channel) || IEEE80211_IS_CHAN_VHT40(channel),
   969→            IEEE80211_IS_CHAN_VHT80(channel));
   970→    }
   971→    apple80211_country_code_data country = {};
