   128→uint16_t ItlIwm::
   129→iwm_scan_rx_chain(struct iwm_softc *sc)
   130→{
   131→    uint16_t rx_chain;
   132→    uint8_t rx_ant;
   133→    
   134→    rx_ant = iwm_fw_valid_rx_ant(sc);
   135→    rx_chain = rx_ant << IWM_PHY_RX_CHAIN_VALID_POS;
   136→    rx_chain |= rx_ant << IWM_PHY_RX_CHAIN_FORCE_MIMO_SEL_POS;
   137→    rx_chain |= rx_ant << IWM_PHY_RX_CHAIN_FORCE_SEL_POS;
   138→    rx_chain |= 0x1 << IWM_PHY_RX_CHAIN_DRIVER_FORCE_POS;
   139→    return htole16(rx_chain);
   140→}

   142→uint32_t ItlIwm::
   143→iwm_scan_rate_n_flags(struct iwm_softc *sc, int flags, int no_cck)
   144→{
   145→    uint32_t tx_ant;
   146→    int i, ind;
   147→    
   148→    for (i = 0, ind = sc->sc_scan_last_antenna;
   149→         i < IWM_RATE_MCS_ANT_NUM; i++) {
   150→        ind = (ind + 1) % IWM_RATE_MCS_ANT_NUM;
   151→        if (iwm_fw_valid_tx_ant(sc) & (1 << ind)) {
   152→            sc->sc_scan_last_antenna = ind;
   153→            break;
   154→        }
   155→    }
   156→    tx_ant = (1 << sc->sc_scan_last_antenna) << IWM_RATE_MCS_ANT_POS;
   157→    
   158→    if ((flags & IEEE80211_CHAN_2GHZ) && !no_cck)
   159→        return htole32(IWL_RATE_1M_PLCP | RATE_MCS_CCK_MSK |
   160→                       tx_ant);
   161→    else
   162→        return htole32(IWL_RATE_6M_PLCP | tx_ant);
   163→}

   165→uint8_t ItlIwm::
   166→iwm_lmac_scan_fill_channels(struct iwm_softc *sc,
   167→                            struct iwm_scan_channel_cfg_lmac *chan, int n_ssids, int bgscan)
   168→{
   169→    struct ieee80211com *ic = &sc->sc_ic;
   170→    struct ieee80211_channel *c;
   171→    uint8_t nchan;
   172→    
   173→    for (nchan = 0, c = &ic->ic_channels[1];
   174→         c <= &ic->ic_channels[IEEE80211_CHAN_MAX] &&
   175→         nchan < sc->sc_capa_n_scan_channels;
   176→         c++) {
   177→        if (c->ic_flags == 0)
   178→            continue;
   179→#ifdef AIRPORT_WCL
   180→        if (!AirportWCL::scanChannelSelected(ic->ic_wcl_scan_channels,
   181→                ic->ic_wcl_scan_requested && ic->ic_wcl_scan_restricted,
   182→                ieee80211_chan2ieee(ic, c)))
   183→            continue;
   184→#endif
   185→        
   186→        chan->channel_num = htole16(ieee80211_mhz2ieee(c->ic_freq, 0));
   187→        chan->iter_count = htole16(1);
   188→        chan->iter_interval = 0;
   189→        chan->flags = htole32(IWM_UNIFIED_SCAN_CHANNEL_PARTIAL);
   190→#ifdef AIRPORT_WCL
   191→        if (AirportWCL::probeChannel(n_ssids != 0,
   192→                c->ic_flags & IEEE80211_CHAN_PASSIVE, c->ic_flags & IEEE80211_CHAN_DFS))
   193→#else
   194→        if (n_ssids != 0 && !bgscan)
   195→#endif
   196→            chan->flags |= htole32(1 << 1); /* select SSID 0 */
   197→        chan++;
   198→        nchan++;
   199→    }
   200→    
   201→    return nchan;
   202→}

   242→int ItlIwm::
   243→iwm_fill_probe_req_v1(struct iwm_softc *sc, struct iwm_scan_probe_req_v1 *preq1)
   244→{
   245→    struct iwm_scan_probe_req preq2;
   246→    int err, i;
   247→    
   248→    err = iwm_fill_probe_req(sc, &preq2);
   249→    if (err)
   250→        return err;
   251→    
   252→    preq1->mac_header = preq2.mac_header;
   253→    for (i = 0; i < nitems(preq1->band_data); i++)
   254→        preq1->band_data[i] = preq2.band_data[i];
   255→    preq1->common_data = preq2.common_data;
   256→    memcpy(preq1->buf, preq2.buf, sizeof(preq1->buf));
   257→    return 0;
   258→}

   260→int ItlIwm::
   261→iwm_fill_probe_req(struct iwm_softc *sc, struct iwm_scan_probe_req *preq)
   262→{
   263→    struct ieee80211com *ic = &sc->sc_ic;
   264→    struct _ifnet *ifp = IC2IFP(ic);
   265→    struct ieee80211_frame *wh = (struct ieee80211_frame *)preq->buf;
   266→    struct ieee80211_rateset *rs;
   267→    size_t remain = sizeof(preq->buf);
   268→    uint8_t *frm, *pos;
   269→    
   270→    memset(preq, 0, sizeof(*preq));
   271→    
   272→    if (remain < sizeof(*wh) + 2)
   273→        return ENOBUFS;
   274→    
   275→    /*
   276→     * Build a probe request frame.  Most of the following code is a
   277→     * copy & paste of what is done in net80211.
   278→     */
   279→    wh->i_fc[0] = IEEE80211_FC0_VERSION_0 | IEEE80211_FC0_TYPE_MGT |
   280→    IEEE80211_FC0_SUBTYPE_PROBE_REQ;
   281→    wh->i_fc[1] = IEEE80211_FC1_DIR_NODS;
   282→    //        IEEE80211_ADDR_COPY(ic->ic_myaddr, LLADDR(ifp->if_sadl));
   283→    IEEE80211_ADDR_COPY(wh->i_addr1, etherbroadcastaddr);
   284→    IEEE80211_ADDR_COPY(wh->i_addr2, ic->ic_myaddr);
   285→    IEEE80211_ADDR_COPY(wh->i_addr3, etherbroadcastaddr);
   286→    *(uint16_t *)&wh->i_dur[0] = 0;    /* filled by HW */
   287→    *(uint16_t *)&wh->i_seq[0] = 0;    /* filled by HW */
   288→    
   289→    frm = (uint8_t *)(wh + 1);
   290→    *frm++ = IEEE80211_ELEMID_SSID;
   291→    *frm++ = 0;
   292→    /* hardware inserts SSID */
   293→    
   294→    /* Tell firmware where the MAC header and SSID IE are. */
   295→    preq->mac_header.offset = 0;
   296→    preq->mac_header.len = htole16(frm - (uint8_t *)wh);
   297→    remain -= frm - (uint8_t *)wh;
   298→    
   299→    /* Fill in 2GHz IEs and tell firmware where they are. */
   300→    rs = &ic->ic_sup_rates[IEEE80211_MODE_11G];
   301→    if (rs->rs_nrates > IEEE80211_RATE_SIZE) {
   302→        if (remain < 4 + rs->rs_nrates)
   303→            return ENOBUFS;
   304→    } else if (remain < 2 + rs->rs_nrates)
   305→        return ENOBUFS;
   306→    preq->band_data[0].offset = htole16(frm - (uint8_t *)wh);
   307→    pos = frm;
   308→    frm = ieee80211_add_rates(frm, rs);
   309→    if (rs->rs_nrates > IEEE80211_RATE_SIZE)
   310→        frm = ieee80211_add_xrates(frm, rs);
   311→    remain -= frm - pos;
   312→    
   313→    if (isset(sc->sc_enabled_capa,
   314→              IWM_UCODE_TLV_CAPA_DS_PARAM_SET_IE_SUPPORT)) {
   315→        if (remain < 3)
   316→            return ENOBUFS;
   317→        *frm++ = IEEE80211_ELEMID_DSPARMS;
   318→        *frm++ = 1;
   319→        *frm++ = 0;
   320→        remain -= 3;
   321→    }
   322→    preq->band_data[0].len = htole16(frm - pos);
   323→    
   324→    if (sc->sc_nvm.sku_cap_band_52GHz_enable) {
   325→        /* Fill in 5GHz IEs. */
   326→        rs = &ic->ic_sup_rates[IEEE80211_MODE_11A];
   327→        if (rs->rs_nrates > IEEE80211_RATE_SIZE) {
   328→            if (remain < 4 + rs->rs_nrates)
   329→                return ENOBUFS;
   330→        } else if (remain < 2 + rs->rs_nrates)
   331→            return ENOBUFS;
   332→        preq->band_data[1].offset = htole16(frm - (uint8_t *)wh);
   333→        pos = frm;
   334→        frm = ieee80211_add_rates(frm, rs);
   335→        if (rs->rs_nrates > IEEE80211_RATE_SIZE)
   336→            frm = ieee80211_add_xrates(frm, rs);
   337→        preq->band_data[1].len = htole16(frm - pos);
   338→        remain -= frm - pos;
   339→        if (ic->ic_flags & IEEE80211_F_VHTON) {
   340→            if (remain < sizeof(struct ieee80211_ie_vhtcap))
   341→                return ENOBUFS;
   342→            frm = ieee80211_add_vhtcaps(frm, ic);
   343→            remain -= frm - pos;
   344→        }
   345→    }
   346→    
   347→    /* Send 11n IEs on both 2GHz and 5GHz bands. */
   348→    preq->common_data.offset = htole16(frm - (uint8_t *)wh);
   349→    pos = frm;
   350→    if (ic->ic_flags & IEEE80211_F_HTON) {
   351→        if (remain < sizeof(struct ieee80211_ie_htcap))
   352→            return ENOBUFS;
   353→        frm = ieee80211_add_htcaps(frm, ic);
   354→        /* XXX add WME info? */
   355→    }
   356→
   357→    preq->common_data.len = htole16(frm - pos);
   358→    
   359→    return 0;
   360→}

   362→int ItlIwm::
   363→iwm_lmac_scan(struct iwm_softc *sc, int bgscan)
   364→{
   365→    struct ieee80211com *ic = &sc->sc_ic;
   366→    int scanSSIDLength = ic->ic_des_esslen;
   367→    const uint8_t *scanSSID = ic->ic_des_essid;
   368→    bool scanActive = scanSSIDLength != 0;
   369→#ifdef AIRPORT_WCL
   370→    scanActive = AirportWCL::scanUsesProbes(ic->ic_wcl_scan_requested,
   371→        ic->ic_wcl_scan_active);
   372→    if (ic->ic_wcl_scan_requested) {
   373→        scanSSIDLength = ic->ic_wcl_scan_ssid_length;
   374→        scanSSID = ic->ic_wcl_scan_ssid;
   375→    }
   376→#endif
   377→
   378→    struct iwm_host_cmd hcmd = {
   379→        .id = IWM_SCAN_OFFLOAD_REQUEST_CMD,
   380→        .len = { 0, },
   381→        .data = { NULL, },
   382→        .flags = 0,
   383→    };
   384→    struct iwm_scan_req_lmac *req;
   385→    struct iwm_scan_probe_req_v1 *preq;
   386→    size_t req_len;
   387→    int err, async = bgscan;
   388→#ifdef AIRPORT_WCL
   389→    const auto timing = AirportWCL::scanTiming(ic->ic_wcl_scan_away_time_ms, bgscan,
   390→        isset(sc->sc_ucode_api, IWM_UCODE_TLV_API_FRAGMENTED_SCAN));
   391→#endif
   392→    
   393→    req_len = sizeof(struct iwm_scan_req_lmac) +
   394→    (sizeof(struct iwm_scan_channel_cfg_lmac) *
   395→     sc->sc_capa_n_scan_channels) + sizeof(struct iwm_scan_probe_req_v1);
   396→    if (req_len > IWM_MAX_CMD_PAYLOAD_SIZE)
   397→        return ENOMEM;
   398→    req = (struct iwm_scan_req_lmac*)malloc(req_len, M_DEVBUF, M_NOWAIT);
   399→    if (req == NULL)
   400→        return ENOMEM;
   401→    bzero(req, req_len);
   402→    
   403→    hcmd.len[0] = (uint16_t)req_len;
   404→    hcmd.data[0] = (void *)req;
   405→    hcmd.flags |= async ? IWM_CMD_ASYNC : 0;
   406→    
   407→    /* These timings correspond to iwlwifi's UNASSOC scan. */
   408→    req->active_dwell = 10;
   409→    req->passive_dwell = 110;
   410→    req->fragmented_dwell = 44;
   411→    req->extended_dwell = 90;
   412→    if (bgscan) {
   413→        req->max_out_time = htole32(120);
   414→        req->suspend_time = htole32(120);
   415→    } else {
   416→        req->max_out_time = htole32(0);
   417→        req->suspend_time = htole32(0);
   418→    }
   419→#ifdef AIRPORT_WCL
   420→    req->max_out_time = htole32(timing.maxOutTU);
   421→    req->suspend_time = htole32(timing.suspendTU);
   422→#endif
   423→    req->scan_prio = htole32(IWM_SCAN_PRIORITY_HIGH);
   424→    req->rx_chain_select = iwm_scan_rx_chain(sc);
   425→    req->iter_num = htole32(1);
   426→    req->delay = 0;
   427→    
   428→    req->scan_flags = htole32(IWM_LMAC_SCAN_FLAG_PASS_ALL |
   429→                              IWM_LMAC_SCAN_FLAG_ITER_COMPLETE |
   430→                              IWM_LMAC_SCAN_FLAG_EXTENDED_DWELL);
   431→#ifdef AIRPORT_WCL
   432→    if (timing.fragmented) {
   433→        req->scan_flags |= htole32(IWM_LMAC_SCAN_FLAG_FRAGMENTED);
   434→        req->scan_flags &= ~htole32(IWM_LMAC_SCAN_FLAG_EXTENDED_DWELL);
   435→    }
   436→    if (scanActive && scanSSIDLength == 0)
   437→        req->scan_flags &= ~htole32(IWM_LMAC_SCAN_FLAG_EXTENDED_DWELL);
   438→#endif
   439→    if (!scanActive)
   440→        req->scan_flags |= htole32(IWM_LMAC_SCAN_FLAG_PASSIVE);
   441→    else if (scanSSIDLength != 0)
   442→        req->scan_flags |=
   443→        htole32(IWM_LMAC_SCAN_FLAG_PRE_CONNECTION);
   444→    if (isset(sc->sc_enabled_capa,
   445→              IWM_UCODE_TLV_CAPA_DS_PARAM_SET_IE_SUPPORT) &&
   446→              isset(sc->sc_enabled_capa,
   447→              IWM_UCODE_TLV_CAPA_WFA_TPC_REP_IE_SUPPORT))
   448→        req->scan_flags |= htole32(IWM_LMAC_SCAN_FLAGS_RRM_ENABLED);
   449→    
   450→    req->flags = htole32(IWM_PHY_BAND_24);
   451→    if (sc->sc_nvm.sku_cap_band_52GHz_enable)
   452→        req->flags |= htole32(IWM_PHY_BAND_5);
   453→    req->filter_flags =
   454→    htole32(IWM_MAC_FILTER_ACCEPT_GRP | IWM_MAC_FILTER_IN_BEACON);
   455→    
   456→    /* Tx flags 2 GHz. */
   457→    req->tx_cmd[0].tx_flags = htole32(IWM_TX_CMD_FLG_SEQ_CTL |
   458→                                      IWM_TX_CMD_FLG_BT_DIS);
   459→    req->tx_cmd[0].rate_n_flags =
   460→    iwm_scan_rate_n_flags(sc, IEEE80211_CHAN_2GHZ, 1/*XXX*/);
   461→    req->tx_cmd[0].sta_id = IWM_AUX_STA_ID;
   462→    
   463→    /* Tx flags 5 GHz. */
   464→    req->tx_cmd[1].tx_flags = htole32(IWM_TX_CMD_FLG_SEQ_CTL |
   465→                                      IWM_TX_CMD_FLG_BT_DIS);
   466→    req->tx_cmd[1].rate_n_flags =
   467→    iwm_scan_rate_n_flags(sc, IEEE80211_CHAN_5GHZ, 1/*XXX*/);
   468→    req->tx_cmd[1].sta_id = IWM_AUX_STA_ID;
   469→    
   470→    /* An empty SSID in an active request is a wildcard probe. */
   471→    if (scanActive) {
   472→        req->direct_scan[0].id = IEEE80211_ELEMID_SSID;
   473→        req->direct_scan[0].len = scanSSIDLength;
   474→        memcpy(req->direct_scan[0].ssid, scanSSID,
   475→               scanSSIDLength);
   476→    }
   477→    
   478→    req->n_channels = iwm_lmac_scan_fill_channels(sc,
   479→                                                  (struct iwm_scan_channel_cfg_lmac *)req->data,
   480→                                                  scanActive, bgscan);
   481→    
   482→    preq = (struct iwm_scan_probe_req_v1 *)(req->data +
   483→                                            (sizeof(struct iwm_scan_channel_cfg_lmac) *
   484→                                             sc->sc_capa_n_scan_channels));
   485→    err = iwm_fill_probe_req_v1(sc, preq);
   486→    if (err) {
   487→        ::free(req);
   488→        return err;
   489→    }
   490→    
   491→    /* Specify the scan plan: We'll do one iteration. */
   492→    req->schedule[0].iterations = 1;
   493→    req->schedule[0].full_scan_mul = 1;
   494→    
   495→    /* Disable EBS. */
   496→    req->channel_opt[0].non_ebs_ratio = 1;
   497→    req->channel_opt[1].non_ebs_ratio = 1;
   498→    
   499→    err = iwm_send_cmd(sc, &hcmd);
   500→    ::free(req);
   501→    return err;
   502→}

   633→int ItlIwm::
   634→iwm_umac_scan(struct iwm_softc *sc, int bgscan)
   635→{
   636→    XYLog("%s\n", __FUNCTION__);
   637→    struct ieee80211com *ic = &sc->sc_ic;
   638→    int scanSSIDLength = ic->ic_des_esslen;
   639→    const uint8_t *scanSSID = ic->ic_des_essid;
   640→    bool scanActive = scanSSIDLength != 0;
   641→#ifdef AIRPORT_WCL
   642→    scanActive = AirportWCL::scanUsesProbes(ic->ic_wcl_scan_requested,
   643→        ic->ic_wcl_scan_active);
   644→    if (ic->ic_wcl_scan_requested) {
   645→        scanSSIDLength = ic->ic_wcl_scan_ssid_length;
   646→        scanSSID = ic->ic_wcl_scan_ssid;
   647→    }
   648→#endif
   649→
   650→    struct iwm_host_cmd hcmd = {
   651→        .id = iwm_cmd_id(IWM_SCAN_REQ_UMAC, IWM_LONG_GROUP, 0),
   652→        .len = { 0, },
   653→        .data = { NULL, },
   654→        .flags = 0,
   655→    };
   656→    struct iwm_scan_req_umac *req;
   657→    void *cmd_data, *tail_data;
   658→    struct iwm_scan_req_umac_tail_v2 *tail;
   659→    struct iwm_scan_req_umac_tail_v1 *tailv1;
   660→    struct iwm_scan_umac_chan_param *chanparam;
   661→    size_t req_len;
   662→    int err, async = bgscan;
   663→#ifdef AIRPORT_WCL
   664→    const auto timing = AirportWCL::scanTiming(ic->ic_wcl_scan_away_time_ms, bgscan,
   665→        isset(sc->sc_ucode_api, IWM_UCODE_TLV_API_FRAGMENTED_SCAN));
   666→#endif
   667→    
   668→    req_len = iwm_umac_scan_size(sc);
   669→    if ((req_len < IWM_SCAN_REQ_UMAC_SIZE_V1 +
   670→         sizeof(struct iwm_scan_req_umac_tail_v1)) ||
   671→        req_len > IWM_MAX_CMD_PAYLOAD_SIZE)
   672→        return ERANGE;
   673→    req = (struct iwm_scan_req_umac*)malloc(req_len, M_DEVBUF, M_NOWAIT);
   674→    if (req == NULL)
   675→        return ENOMEM;
   676→    bzero(req, req_len);
   677→    
   678→    hcmd.len[0] = (uint16_t)req_len;
   679→    hcmd.data[0] = (void *)req;
   680→    hcmd.flags |= async ? IWM_CMD_ASYNC : 0;
   681→    
   682→    if (isset(sc->sc_ucode_api, IWM_UCODE_TLV_API_ADAPTIVE_DWELL)) {
   683→        req->v7.adwell_default_n_aps_social =
   684→        IWM_SCAN_ADWELL_DEFAULT_N_APS_SOCIAL;
   685→        req->v7.adwell_default_n_aps =
   686→        IWM_SCAN_ADWELL_DEFAULT_LB_N_APS;
   687→        
   688→        if (scanSSIDLength != 0)
   689→            req->v7.adwell_max_budget =
   690→            htole16(IWM_SCAN_ADWELL_MAX_BUDGET_DIRECTED_SCAN);
   691→        else
   692→            req->v7.adwell_max_budget =
   693→            htole16(IWM_SCAN_ADWELL_MAX_BUDGET_FULL_SCAN);
   694→        
   695→        req->v7.scan_priority = htole32(IWM_SCAN_PRIORITY_HIGH);
   696→        req->v7.max_out_time[IWM_SCAN_LB_LMAC_IDX] = 0;
   697→        req->v7.suspend_time[IWM_SCAN_LB_LMAC_IDX] = 0;
   698→        
   699→        if (isset(sc->sc_ucode_api,
   700→                  IWM_UCODE_TLV_API_ADAPTIVE_DWELL_V2)) {
   701→            req->v8.active_dwell[IWM_SCAN_LB_LMAC_IDX] = 10;
   702→            req->v8.passive_dwell[IWM_SCAN_LB_LMAC_IDX] = 110;
   703→        } else {
   704→            req->v7.active_dwell = 10;
   705→            req->v7.passive_dwell = 110;
   706→            req->v7.fragmented_dwell = 44;
   707→        }
   708→    } else {
   709→        /* These timings correspond to iwlwifi's UNASSOC scan. */
   710→        req->v1.active_dwell = 10;
   711→        req->v1.passive_dwell = 110;
   712→        req->v1.fragmented_dwell = 44;
   713→        req->v1.extended_dwell = 90;
   714→
   715→        req->v1.scan_priority = htole32(IWM_SCAN_PRIORITY_HIGH);
   716→    }
   717→    
   718→    if (bgscan) {
   719→#ifdef AIRPORT_WCL
   720→        const uint32_t timeout = htole32(timing.suspendTU);
   721→        const uint32_t maxOut = htole32(timing.maxOutTU);
   722→#else
   723→        const uint32_t timeout = htole32(120);
   724→        const uint32_t maxOut = timeout;
   725→#endif
   726→        if (isset(sc->sc_ucode_api,
   727→                  IWM_UCODE_TLV_API_ADAPTIVE_DWELL_V2)) {
   728→            req->v8.max_out_time[IWM_SCAN_LB_LMAC_IDX] = maxOut;
   729→            req->v8.suspend_time[IWM_SCAN_LB_LMAC_IDX] = timeout;
   730→        } else if (isset(sc->sc_ucode_api,
   731→                         IWM_UCODE_TLV_API_ADAPTIVE_DWELL)) {
   732→            req->v7.max_out_time[IWM_SCAN_LB_LMAC_IDX] = maxOut;
   733→            req->v7.suspend_time[IWM_SCAN_LB_LMAC_IDX] = timeout;
   734→        } else {
   735→            req->v1.max_out_time = maxOut;
   736→            req->v1.suspend_time = timeout;
   737→        }
   738→    }
   739→
   740→    req->ooc_priority = htole32(IWM_SCAN_PRIORITY_HIGH);
   741→    
   742→    cmd_data = iwm_get_scan_req_umac_data(sc, req);
   743→    chanparam = iwm_get_scan_req_umac_chan_param(sc, req);
   744→    chanparam->count = iwm_umac_scan_fill_channels(sc,
   745→                                                   (struct iwm_scan_channel_cfg_umac *)cmd_data,
   746→                                                   scanActive, bgscan);
   747→    chanparam->flags = 0;
   748→    
   749→    tail_data = (uint8_t*)cmd_data + sizeof(struct iwm_scan_channel_cfg_umac) *
   750→    sc->sc_capa_n_scan_channels;
   751→    tail = (struct iwm_scan_req_umac_tail_v2*)((uint8_t*)tail_data);
   752→    /* tail v1 layout differs in preq and direct_scan member fields. */
   753→    tailv1 = (struct iwm_scan_req_umac_tail_v1*)((uint8_t*)tail_data);
   754→    
   755→    req->general_flags = htole32(IWM_UMAC_SCAN_GEN_FLAGS_PASS_ALL |
   756→                                 IWM_UMAC_SCAN_GEN_FLAGS_ITER_COMPLETE);
   757→    if (isset(sc->sc_ucode_api, IWM_UCODE_TLV_API_ADAPTIVE_DWELL_V2)) {
   758→        req->v8.general_flags2 =
   759→        IWM_UMAC_SCAN_GEN_FLAGS2_ALLOW_CHNL_REORDER;
   760→    }
   761→    
   762→    /* An empty SSID in an active request is a wildcard probe. */
   763→    if (scanActive) {
   764→        if (isset(sc->sc_ucode_api, IWM_UCODE_TLV_API_SCAN_EXT_CHAN_VER)) {
   765→            tail->direct_scan[0].id = IEEE80211_ELEMID_SSID;
   766→            tail->direct_scan[0].len = scanSSIDLength;
   767→            memcpy(tail->direct_scan[0].ssid, scanSSID,
   768→                   scanSSIDLength);
   769→        } else {
   770→            tailv1->direct_scan[0].id = IEEE80211_ELEMID_SSID;
   771→            tailv1->direct_scan[0].len = scanSSIDLength;
   772→            memcpy(tailv1->direct_scan[0].ssid, scanSSID,
   773→                   scanSSIDLength);
   774→        }
   775→        if (scanSSIDLength != 0)
   776→            req->general_flags |= htole32(IWM_UMAC_SCAN_GEN_FLAGS_PRE_CONNECT);
   777→    } else
   778→        req->general_flags |= htole32(IWM_UMAC_SCAN_GEN_FLAGS_PASSIVE);
   779→    
   780→    if (isset(sc->sc_enabled_capa,
   781→              IWM_UCODE_TLV_CAPA_DS_PARAM_SET_IE_SUPPORT) &&
   782→              isset(sc->sc_enabled_capa,
   783→              IWM_UCODE_TLV_CAPA_WFA_TPC_REP_IE_SUPPORT))
   784→        req->general_flags |=
   785→        htole32(IWM_UMAC_SCAN_GEN_FLAGS_RRM_ENABLED);
   786→    
   787→    if (isset(sc->sc_ucode_api, IWM_UCODE_TLV_API_ADAPTIVE_DWELL)) {
   788→        req->general_flags |=
   789→        htole32(IWM_UMAC_SCAN_GEN_FLAGS_ADAPTIVE_DWELL);
   790→    } else {
   791→        req->general_flags |=
   792→        htole32(IWM_UMAC_SCAN_GEN_FLAGS_EXTENDED_DWELL);
   793→    }
   794→    
   795→#ifdef AIRPORT_WCL
   796→    if (timing.fragmented) {
   797→        req->general_flags &= ~htole16(IWM_UMAC_SCAN_GEN_FLAGS_EXTENDED_DWELL);
   798→        req->general_flags |= htole16(IWM_UMAC_SCAN_GEN_FLAGS_FRAGMENTED);
   799→        if (isset(sc->sc_ucode_api, IWM_UCODE_TLV_API_ADAPTIVE_DWELL_V2))
   800→            req->v8.num_of_fragments[IWM_SCAN_LB_LMAC_IDX] = 3;
   801→    }
   802→#endif
   803→
   804→    if (isset(sc->sc_ucode_api, IWM_UCODE_TLV_API_SCAN_EXT_CHAN_VER))
   805→        err = iwm_fill_probe_req(sc, &tail->preq);
   806→    else
   807→        err = iwm_fill_probe_req_v1(sc, &tailv1->preq);
   808→    if (err) {
   809→        ::free(req);
   810→        return err;
   811→    }
   812→    
   813→    /* Specify the scan plan: We'll do one iteration. */
   814→    tail->schedule[0].interval = 0;
   815→    tail->schedule[0].iter_count = 1;
   816→    
   817→    err = iwm_send_cmd(sc, &hcmd);
   818→    ::free(req);
   819→    return err;
   820→}

   843→uint8_t ItlIwm::
   844→iwm_ridx2rate(struct ieee80211_rateset *rs, int ridx)
   845→{
   846→    int i;
   847→    uint8_t rval;
   848→    
   849→    for (i = 0; i < rs->rs_nrates; i++) {
   850→        rval = (rs->rs_rates[i] & IEEE80211_RATE_VAL);
   851→        if (rval == ieee80211_std_rateset_11g.rs_rates[ridx])
   852→            return rs->rs_rates[i];
   853→    }
   854→    
   855→    return 0;

   857→
   858→void ItlIwm::
   859→iwm_ack_rates(struct iwm_softc *sc, struct iwm_node *in, int *cck_rates,
   860→              int *ofdm_rates)
   861→{
   862→    struct ieee80211_node *ni = &in->in_ni;
   863→    struct ieee80211_rateset *rs = &ni->ni_rates;
   864→    int lowest_present_ofdm = -1;
   865→    int lowest_present_cck = -1;
   866→    uint8_t cck = 0;
   867→    uint8_t ofdm = 0;
   868→    int i;
   869→    
   870→    if (ni->ni_chan == IEEE80211_CHAN_ANYC ||
   871→        IEEE80211_IS_CHAN_2GHZ(ni->ni_chan)) {
   872→        for (i = IWL_FIRST_CCK_RATE; i < IWL_FIRST_OFDM_RATE; i++) {
   873→            if ((iwm_ridx2rate(rs, i) & IEEE80211_RATE_BASIC) == 0)
   874→                continue;
   875→            cck |= (1 << i);
   876→            if (lowest_present_cck == -1 || lowest_present_cck > i)
   877→                lowest_present_cck = i;
   878→        }
   879→    }
   880→    for (i = IWL_FIRST_OFDM_RATE; i <= IWL_LAST_NON_HT_RATE; i++) {
   881→        if ((iwm_ridx2rate(rs, i) & IEEE80211_RATE_BASIC) == 0)
   882→            continue;
   883→        ofdm |= (1 << (i - IWL_FIRST_OFDM_RATE));
   884→        if (lowest_present_ofdm == -1 || lowest_present_ofdm > i)
   885→            lowest_present_ofdm = i;
   886→    }
   887→    
   888→    /*
   889→     * Now we've got the basic rates as bitmaps in the ofdm and cck
   890→     * variables. This isn't sufficient though, as there might not
   891→     * be all the right rates in the bitmap. E.g. if the only basic
   892→     * rates are 5.5 Mbps and 11 Mbps, we still need to add 1 Mbps
   893→     * and 6 Mbps because the 802.11-2007 standard says in 9.6:
   894→     *
   895→     *    [...] a STA responding to a received frame shall transmit
   896→     *    its Control Response frame [...] at the highest rate in the
   897→     *    BSSBasicRateSet parameter that is less than or equal to the
   898→     *    rate of the immediately previous frame in the frame exchange
   899→     *    sequence ([...]) and that is of the same modulation class
   900→     *    ([...]) as the received frame. If no rate contained in the
   901→     *    BSSBasicRateSet parameter meets these conditions, then the
   902→     *    control frame sent in response to a received frame shall be
   903→     *    transmitted at the highest mandatory rate of the PHY that is
   904→     *    less than or equal to the rate of the received frame, and
   905→     *    that is of the same modulation class as the received frame.
   906→     *
   907→     * As a consequence, we need to add all mandatory rates that are
   908→     * lower than all of the basic rates to these bitmaps.
   909→     */
   910→    
   911→    if (IWL_RATE_24M_INDEX < lowest_present_ofdm)
   912→        ofdm |= IWL_RATE_BIT_MSK(24) >> IWL_FIRST_OFDM_RATE;
   913→    if (IWL_RATE_12M_INDEX < lowest_present_ofdm)
   914→        ofdm |= IWL_RATE_BIT_MSK(12) >> IWL_FIRST_OFDM_RATE;
   915→    /* 6M already there or needed so always add */
   916→    ofdm |= IWL_RATE_BIT_MSK(6) >> IWL_FIRST_OFDM_RATE;
   917→    
   918→    /*
   919→     * CCK is a bit more complex with DSSS vs. HR/DSSS vs. ERP.
   920→     * Note, however:
   921→     *  - if no CCK rates are basic, it must be ERP since there must
   922→     *    be some basic rates at all, so they're OFDM => ERP PHY
   923→     *    (or we're in 5 GHz, and the cck bitmap will never be used)
   924→     *  - if 11M is a basic rate, it must be ERP as well, so add 5.5M
   925→     *  - if 5.5M is basic, 1M and 2M are mandatory
   926→     *  - if 2M is basic, 1M is mandatory
   927→     *  - if 1M is basic, that's the only valid ACK rate.
   928→     * As a consequence, it's not as complicated as it sounds, just add
   929→     * any lower rates to the ACK rate bitmap.
   930→     */
   931→    if (IWL_RATE_11M_INDEX < lowest_present_cck)
   932→        cck |= IWL_RATE_BIT_MSK(11) >> IWL_FIRST_CCK_RATE;
   933→    if (IWL_RATE_5M_INDEX < lowest_present_cck)
   934→        cck |= IWL_RATE_BIT_MSK(5) >> IWL_FIRST_CCK_RATE;
   935→    if (IWL_RATE_2M_INDEX < lowest_present_cck)
   936→        cck |= IWL_RATE_BIT_MSK(2) >> IWL_FIRST_CCK_RATE;
   937→    /* 1M already there or needed so always add */
   938→    cck |= IWL_RATE_BIT_MSK(1) >> IWL_FIRST_CCK_RATE;
   939→    
   940→    *cck_rates = cck;
   941→    *ofdm_rates = ofdm;
   942→}

   944→int ItlIwm::
   945→iwm_scan(struct iwm_softc *sc)
   946→{
   947→    XYLog("%s\n", __FUNCTION__);
   948→    struct ieee80211com *ic = &sc->sc_ic;
   949→    struct _ifnet *ifp = IC2IFP(ic);
   950→    int err;
   951→    
   952→    if (sc->sc_flags & IWM_FLAG_BGSCAN) {
   953→        err = iwm_scan_abort(sc);
   954→        if (err) {
   955→            XYLog("%s: could not abort background scan\n",
   956→                  DEVNAME(sc));
   957→            return err;
   958→        }
   959→    }
   960→    
   961→    if (isset(sc->sc_enabled_capa, IWM_UCODE_TLV_CAPA_UMAC_SCAN))
   962→        err = iwm_umac_scan(sc, 0);
   963→    else
   964→        err = iwm_lmac_scan(sc, 0);
   965→    if (err && err != 1) {
   966→        XYLog("%s: %d could not initiate scan, err=%d\n", DEVNAME(sc), __LINE__, err);
   967→        return err;
   968→    }
   969→    
   970→    /*
   971→     * The current mode might have been fixed during association.
   972→     * Ensure all channels get scanned.
   973→     */
   974→    if (IFM_MODE(ic->ic_media.ifm_cur->ifm_media) == IFM_AUTO)
   975→        ieee80211_setmode(ic, IEEE80211_MODE_AUTO);
   976→    
   977→    sc->sc_flags |= IWM_FLAG_SCANNING;
   978→    if (ifp->if_flags & IFF_DEBUG)
   979→        XYLog("%s: %s -> %s\n", ifp->if_xname,
   980→              ieee80211_state_name[ic->ic_state],
   981→              ieee80211_state_name[IEEE80211_S_SCAN]);
   982→    if ((sc->sc_flags & IWM_FLAG_BGSCAN) == 0) {
   983→        ieee80211_set_link_state(ic, LINK_STATE_DOWN);
   984→        ieee80211_node_cleanup(ic, ic->ic_bss);
   985→    }
   986→    ic->ic_state = IEEE80211_S_SCAN;
   987→    iwm_led_blink_start(sc);
   988→    wakeupOn(&ic->ic_state); /* wake iwm_init() */
   989→    
   990→    return 0;
   991→}

   993→int ItlIwm::
   994→iwm_bgscan(struct ieee80211com *ic)
   995→{
   996→    XYLog("%s\n", __FUNCTION__);
   997→    struct iwm_softc *sc = (struct iwm_softc *)IC2IFP(ic)->if_softc;
   998→    ItlIwm *that = container_of(sc, ItlIwm, com);
   999→    int err;
  1000→    
  1001→    if (sc->sc_flags & IWM_FLAG_SCANNING)
  1002→        return 0;
  1003→    
  1004→    if (isset(sc->sc_enabled_capa, IWM_UCODE_TLV_CAPA_UMAC_SCAN))
  1005→        err = that->iwm_umac_scan(sc, 1);
  1006→    else
  1007→        err = that->iwm_lmac_scan(sc, 1);
  1008→    if (err && err != 1) {
  1009→        XYLog("%s: could not initiate scan\n", DEVNAME(sc));
  1010→        return err;
  1011→    }
  1012→    
  1013→    sc->sc_flags |= IWM_FLAG_BGSCAN;
  1014→    return 0;
  1015→}

  1017→int ItlIwm::
  1018→iwm_umac_scan_abort(struct iwm_softc *sc)
  1019→{
  1020→    struct iwm_umac_scan_abort cmd = { 0 };
  1021→    
  1022→    return iwm_send_cmd_pdu(sc,
  1023→                            IWM_WIDE_ID(IWM_LONG_GROUP, IWM_SCAN_ABORT_UMAC),
  1024→                            0, sizeof(cmd), &cmd);
  1025→}

  1027→int ItlIwm::
  1028→iwm_lmac_scan_abort(struct iwm_softc *sc)
  1029→{
  1030→    struct iwm_host_cmd cmd = {
  1031→        .id = IWM_SCAN_OFFLOAD_ABORT_CMD,
  1032→    };
  1033→    int err;
  1034→    uint32_t status;
  1035→    
  1036→    err = iwm_send_cmd_status(sc, &cmd, &status);
  1037→    if (err)
  1038→        return err;
  1039→    
  1040→    if (status != IWM_CAN_ABORT_STATUS) {
  1041→        /*
  1042→         * The scan abort will return 1 for success or
  1043→         * 2 for "failure".  A failure condition can be
  1044→         * due to simply not being in an active scan which
  1045→         * can occur if we send the scan abort before the
  1046→         * microcode has notified us that a scan is completed.
  1047→         */
  1048→        return EBUSY;
  1049→    }
  1050→    
  1051→    return 0;
  1052→}

  1054→int ItlIwm::
  1055→iwm_scan_abort(struct iwm_softc *sc)
  1056→{
  1057→    int err;
  1058→    
  1059→    if (isset(sc->sc_enabled_capa, IWM_UCODE_TLV_CAPA_UMAC_SCAN))
  1060→        err = iwm_umac_scan_abort(sc);
  1061→    else
  1062→        err = iwm_lmac_scan_abort(sc);
  1063→    
  1064→    if (err == 0)
  1065→        sc->sc_flags &= ~(IWM_FLAG_SCANNING | IWM_FLAG_BGSCAN);
  1066→    return err;
  1067→}
