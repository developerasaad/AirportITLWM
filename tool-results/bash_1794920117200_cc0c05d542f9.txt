   216→IOReturn AirportItlwm::
   217→getSSID(OSObject *object,
   218→                        struct apple80211_ssid_data *sd)
   219→{
   220→    struct ieee80211com * ic = fHalService->get80211Controller();
   221→    if (ic->ic_state == IEEE80211_S_RUN) {
   222→        memset(sd, 0, sizeof(*sd));
   223→        sd->version = APPLE80211_VERSION;
   224→        memcpy(sd->ssid_bytes, ic->ic_des_essid, strlen((const char*)ic->ic_des_essid));
   225→        sd->ssid_len = (uint32_t)strlen((const char*)ic->ic_des_essid);
   226→        return kIOReturnSuccess;
   227→    }
   228→    return 6;
   229→}

   254→IOReturn AirportItlwm::
   255→setCIPHER_KEY(OSObject *object, struct apple80211_key *key)
   256→{
   257→    XYLog("%s\n", __FUNCTION__);
   258→    const char* keydump = hexdump(key->key, key->key_len);
   259→    const char* rscdump = hexdump(key->key_rsc, key->key_rsc_len);
   260→    const char* eadump = hexdump(key->key_ea.octet, APPLE80211_ADDR_LEN);
   261→    static_assert(__offsetof(struct apple80211_key, key_ea) == 92, "struct corrupted");
   262→    static_assert(__offsetof(struct apple80211_key, key_rsc_len) == 80, "struct corrupted");
   263→    static_assert(__offsetof(struct apple80211_key, wowl_kck_len) == 100, "struct corrupted");
   264→    static_assert(__offsetof(struct apple80211_key, wowl_kek_len) == 120, "struct corrupted");
   265→    static_assert(__offsetof(struct apple80211_key, wowl_kck_key) == 104, "struct corrupted");
   266→    if (keydump && rscdump && eadump)
   267→        XYLog("Set key request: len=%d cipher_type=%d flags=%d index=%d key=%s rsc_len=%d rsc=%s ea=%s\n",
   268→              key->key_len, key->key_cipher_type, key->key_flags, key->key_index, keydump, key->key_rsc_len, rscdump, eadump);
   269→    else
   270→        XYLog("Set key request, but failed to allocate memory for hexdump\n");
   271→    
   272→    if (keydump)
   273→        IOFree((void*)keydump, 3 * key->key_len + 1);
   274→    if (rscdump)
   275→        IOFree((void*)rscdump, 3 * key->key_rsc_len + 1);
   276→    if (eadump)
   277→        IOFree((void*)eadump, 3 * APPLE80211_ADDR_LEN + 1);
   278→    
   279→    switch (key->key_cipher_type) {
   280→        case APPLE80211_CIPHER_NONE:
   281→            // clear existing key
   282→//            XYLog("Setting NONE key is not supported\n");
   283→            break;
   284→        case APPLE80211_CIPHER_WEP_40:
   285→        case APPLE80211_CIPHER_WEP_104:
   286→            XYLog("Setting WEP key %d is not supported\n", key->key_index);
   287→            break;
   288→        case APPLE80211_CIPHER_TKIP:
   289→        case APPLE80211_CIPHER_AES_OCB:
   290→        case APPLE80211_CIPHER_AES_CCM:
   291→            switch (key->key_flags) {
   292→                case 4: // PTK
   293→                    setPTK(key->key, key->key_len);
   294→                    getNetworkInterface()->postMessage(APPLE80211_M_RSN_HANDSHAKE_DONE);
   295→                    break;
   296→                case 0: // GTK
   297→                    setGTK(key->key, key->key_len, key->key_index, key->key_rsc);
   298→                    getNetworkInterface()->postMessage(APPLE80211_M_RSN_HANDSHAKE_DONE);
   299→                    break;
   300→            }
   301→            break;
   302→        case APPLE80211_CIPHER_PMK:
   303→            XYLog("Setting WPA PMK is not supported\n");
   304→            break;
   305→        case APPLE80211_CIPHER_PMKSA:
   306→            XYLog("Setting WPA PMKSA is not supported\n");
   307→            break;
   308→    }
   309→    //fInterface->postMessage(APPLE80211_M_CIPHER_KEY_CHANGED);
   310→    return kIOReturnSuccess;
   311→}

   313→// From Ventura, airport/wifiagent seems that they don't like to accept extra channel flags in the scan result list,
   314→// if not, the exact behavior is that the wifi list on the control center/menu bar will not refresh after system boot.
   315→#if __IO80211_TARGET >= __MAC_13_0
   316→static int ieeeChanFlag2appleScanFlagVentura(int flags)
   317→{
   318→    int ret = 0;
   319→    if (flags & IEEE80211_CHAN_2GHZ)
   320→        ret |= APPLE80211_C_FLAG_2GHZ;
   321→    if (flags & IEEE80211_CHAN_5GHZ)
   322→        ret |= APPLE80211_C_FLAG_5GHZ;
   323→    ret |= (APPLE80211_C_FLAG_ACTIVE | APPLE80211_C_FLAG_20MHZ);
   324→    return ret;
   325→}
   326→#endif
   327→
   328→static int ieeeChanFlag2apple(int flags, int bw)
   329→{
   330→    int ret = 0;
   331→    if (flags & IEEE80211_CHAN_2GHZ)
   332→        ret |= APPLE80211_C_FLAG_2GHZ;
   333→    if (flags & IEEE80211_CHAN_5GHZ)
   334→        ret |= APPLE80211_C_FLAG_5GHZ;
   335→    if (!(flags & IEEE80211_CHAN_PASSIVE))
   336→        ret |= APPLE80211_C_FLAG_ACTIVE;
   337→    if (flags & IEEE80211_CHAN_DFS)
   338→        ret |= APPLE80211_C_FLAG_DFS;
   339→    if (bw == -1) {
   340→        if (flags & IEEE80211_CHAN_VHT) {
   341→            if ((flags & IEEE80211_CHAN_VHT160) || (flags & IEEE80211_CHAN_VHT80_80))
   342→                ret |= APPLE80211_C_FLAG_160MHZ;
   343→            if (flags & IEEE80211_CHAN_VHT80)
   344→                ret |= APPLE80211_C_FLAG_80MHZ;
   345→        } else if ((flags & IEEE80211_CHAN_HT40) && (flags & IEEE80211_CHAN_HT)) {
   346→            ret |= APPLE80211_C_FLAG_40MHZ;
   347→            if (flags & IEEE80211_CHAN_HT40U)
   348→                ret |= APPLE80211_C_FLAG_EXT_ABV;
   349→        } else if (flags & IEEE80211_CHAN_HT20)
   350→            ret |= APPLE80211_C_FLAG_20MHZ;
   351→        else if ((flags & IEEE80211_CHAN_CCK) || (flags & IEEE80211_CHAN_OFDM))
   352→            ret |= APPLE80211_C_FLAG_10MHZ;
   353→    } else {
   354→        switch (bw) {
   355→            case IEEE80211_CHAN_WIDTH_80P80:
   356→            case IEEE80211_CHAN_WIDTH_160:
   357→                ret |= APPLE80211_C_FLAG_160MHZ;
   358→                break;
   359→            case IEEE80211_CHAN_WIDTH_80:
   360→                ret |= APPLE80211_C_FLAG_80MHZ;
   361→                break;
   362→            case IEEE80211_CHAN_WIDTH_40:
   363→                ret |= APPLE80211_C_FLAG_40MHZ;
   364→                if (flags & IEEE80211_CHAN_HT40U)
   365→                    ret |= APPLE80211_C_FLAG_EXT_ABV;
   366→                break;
   367→            case IEEE80211_CHAN_WIDTH_20:
   368→                ret |= APPLE80211_C_FLAG_20MHZ;
   369→                break;
   370→            default:
   371→                if (flags & IEEE80211_CHAN_HT20)
   372→                    ret |= APPLE80211_C_FLAG_20MHZ;
   373→                else if ((flags & IEEE80211_CHAN_CCK) || (flags & IEEE80211_CHAN_OFDM))
   374→                    ret |= APPLE80211_C_FLAG_10MHZ;
   375→                break;
   376→        }
   377→    }
   378→    return ret;
   379→}

   381→IOReturn AirportItlwm::
   382→getCHANNEL(OSObject *object,
   383→                           struct apple80211_channel_data *cd)
   384→{
   385→    struct ieee80211com * ic = fHalService->get80211Controller();
   386→    if (ic->ic_state == IEEE80211_S_RUN) {
   387→        memset(cd, 0, sizeof(apple80211_channel_data));
   388→        cd->version = APPLE80211_VERSION;
   389→        cd->channel.version = APPLE80211_VERSION;
   390→        cd->channel.channel = ieee80211_chan2ieee(ic, ic->ic_bss->ni_chan);
   391→        cd->channel.flags = ieeeChanFlag2apple(ic->ic_bss->ni_chan->ic_flags, ic->ic_bss->ni_chw);
   392→        return kIOReturnSuccess;
   393→    }
   394→    return 6;
   395→}

   470→IOReturn AirportItlwm::
   471→getRATE(OSObject *object, struct apple80211_rate_data *rd)
   472→{
   473→    struct ieee80211com *ic = fHalService->get80211Controller();
   474→    if (ic->ic_bss == NULL)
   475→        return 6;
   476→    int nss;
   477→    int sgi;
   478→    int index = 0;
   479→    if (ic->ic_state == IEEE80211_S_RUN) {
   480→        memset(rd, 0, sizeof(*rd));
   481→        rd->version = APPLE80211_VERSION;
   482→        rd->num_radios = 1;
   483→        sgi = ieee80211_node_supports_sgi(ic->ic_bss);
   484→        if (ic->ic_curmode == IEEE80211_MODE_11AC) {
   485→            if (sgi)
   486→                index += 1;
   487→            nss = fHalService->getDriverInfo()->getTxNSS();
   488→            switch (ic->ic_bss->ni_chw) {
   489→                case IEEE80211_CHAN_WIDTH_40:
   490→                    index += 4;
   491→                    break;
   492→                case IEEE80211_CHAN_WIDTH_80:
   493→                    index += 8;
   494→                    break;
   495→                case IEEE80211_CHAN_WIDTH_80P80:
   496→                case IEEE80211_CHAN_WIDTH_160:
   497→                    index += 12;
   498→                    break;
   499→
   500→                default:
   501→                    break;
   502→            }
   503→            index += 2 * (nss - 1);
   504→            const struct ieee80211_vht_rateset *rs = &ieee80211_std_ratesets_11ac[index];
   505→            rd->rate[0] = rs->rates[ic->ic_bss->ni_txmcs % rs->nrates] / 2;
   506→        } else if (ic->ic_curmode == IEEE80211_MODE_11N) {
   507→            int is_40mhz = ic->ic_bss->ni_chw == IEEE80211_CHAN_WIDTH_40;
   508→            if (sgi)
   509→                index += 1;
   510→            if (is_40mhz)
   511→                index += (IEEE80211_HT_RATESET_MIMO4_SGI + 1);
   512→            index += (ic->ic_bss->ni_txmcs / 16);
   513→            nss = ic->ic_bss->ni_txmcs / 8 + 1;
   514→            index += 2 * (nss - 1);
   515→            rd->rate[0] = ieee80211_std_ratesets_11n[index].rates[ic->ic_bss->ni_txmcs % 8] / 2;
   516→        } else
   517→            rd->rate[0] = ic->ic_bss->ni_rates.rs_rates[ic->ic_bss->ni_txrate];
   518→        return kIOReturnSuccess;
   519→    }
   520→    return 6;
   521→}

   668→                                     struct apple80211_capability_data *cd)

   670→    uint32_t caps = fHalService->get80211Controller()->ic_caps;
   671→    memset(cd, 0, sizeof(struct apple80211_capability_data));
   672→    
   673→    if (caps & IEEE80211_C_WEP)
   674→        cd->capabilities[0] |= 1 << APPLE80211_CAP_WEP;
   675→    if (caps & IEEE80211_C_RSN)
   676→        cd->capabilities[0] |= 1 << APPLE80211_CAP_TKIP | 1 << APPLE80211_CAP_AES_CCM;
   677→    // Disable not implemented capabilities
   678→    // if (caps & IEEE80211_C_PMGT)

   682→    // if (caps & IEEE80211_C_HOSTAP)
   683→    //     cd->capabilities[0] |= 1 << APPLE80211_CAP_HOSTAP;
   684→    // AES not enabled, like on Apple cards
   685→    
   686→    if (caps & IEEE80211_C_SHSLOT)
   687→        cd->capabilities[1] |= 1 << (APPLE80211_CAP_SHSLOT - 8);
   688→    if (caps & IEEE80211_C_SHPREAMBLE)
   689→        cd->capabilities[1] |= 1 << (APPLE80211_CAP_SHPREAMBLE - 8);
   690→    if (caps & IEEE80211_C_RSN)
   691→        cd->capabilities[1] |= 1 << (APPLE80211_CAP_WPA1 - 8) | 1 << (APPLE80211_CAP_WPA2 - 8) | 1 << (APPLE80211_CAP_TKIPMIC - 8);
   692→    // Disable not implemented capabilities
   693→    // if (caps & IEEE80211_C_TXPMGT)
   694→    //     cd->capabilities[1] |= 1 << (APPLE80211_CAP_TXPMGT - 8);
   695→    // if (caps & IEEE80211_C_MONITOR)
   696→    //     cd->capabilities[1] |= 1 << (APPLE80211_CAP_MONITOR - 8);
   697→    // WPA not enabled, like on Apple cards
   698→
   699→    cd->version = APPLE80211_VERSION;
   700→    cd->capabilities[2] = 0xFF; // BURST, WME, SHORT_GI_40MHZ, SHORT_GI_20MHZ, WOW, TSN, ?, ?
   701→    cd->capabilities[3] = 0x2B;
   702→    cd->capabilities[4] = 0xAD;
   703→    cd->capabilities[5] = 0x80;//isCntryDefaultSupported
   704→    cd->capabilities[5] |= 0x0C;
   705→    cd->capabilities[6] = (
   706→//                           1 |    //MFP capable
   707→                           0x8 |
   708→                           0x4 |

   711→    cd->capabilities[7] = 0x84; // This byte contains Apple Watch unlock

   713→    //cd->capabilities[8] |= 8;//dfs white list
