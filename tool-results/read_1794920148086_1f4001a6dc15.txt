    36→#define  PCI_MSI_FLAGS        2    /* Message Control */
    37→#define  PCI_CAP_ID_MSI        0x05    /* Message Signalled Interrupts */
    38→#define  PCI_MSIX_FLAGS        2    /* Message Control */
    39→#define  PCI_CAP_ID_MSIX    0x11    /* MSI-X */
    40→#define  PCI_MSIX_FLAGS_ENABLE    0x8000    /* MSI-X enable */
    41→#define  PCI_MSI_FLAGS_ENABLE    0x0001    /* MSI feature enabled */
    42→
    43→static void pciMsiSetEnable(IOPCIDevice *device, UInt8 msiCap, int enable)
    44→{
    45→    u16 control;
    46→    
    47→    control = device->configRead16(msiCap + PCI_MSI_FLAGS);
    48→    control &= ~PCI_MSI_FLAGS_ENABLE;
    49→    if (enable)
    50→        control |= PCI_MSI_FLAGS_ENABLE;
    51→    device->configWrite16(msiCap + PCI_MSI_FLAGS, control);
    52→}
    53→
    54→static void pciMsiXClearAndSet(IOPCIDevice *device, UInt8 msixCap, UInt16 clear, UInt16 set)
    55→{
    56→    u16 ctrl;
    57→    
    58→    ctrl = device->configRead16(msixCap + PCI_MSIX_FLAGS);
    59→    ctrl &= ~clear;
    60→    ctrl |= set;
    61→    device->configWrite16(msixCap + PCI_MSIX_FLAGS, ctrl);
    62→}

    64→IOService* AirportItlwm::probe(IOService *provider, SInt32 *score)
    65→{
    66→    bool isMatch = false;
    67→    super::probe(provider, score);
    68→    UInt8 msiCap;
    69→    UInt8 msixCap;
    70→    IOPCIDevice* device = OSDynamicCast(IOPCIDevice, provider);
    71→    if (!device)
    72→        return NULL;
    73→    if (ItlIwx::iwx_match(device)) {
    74→        isMatch = true;
    75→        fHalService = new ItlIwx;
    76→    }
    77→    if (!isMatch && ItlIwm::iwm_match(device)) {
    78→        isMatch = true;
    79→        fHalService = new ItlIwm;
    80→    }
    81→    if (!isMatch && ItlIwn::iwn_match(device)) {
    82→        isMatch = true;
    83→        fHalService = new ItlIwn;
    84→    }
    85→    if (isMatch) {
    86→        device->findPCICapability(PCI_CAP_ID_MSIX, &msixCap);
    87→        if (msixCap)
    88→            pciMsiXClearAndSet(device, msixCap, PCI_MSIX_FLAGS_ENABLE, 0);
    89→        device->findPCICapability(PCI_CAP_ID_MSI, &msiCap);
    90→        if (msiCap)
    91→            pciMsiSetEnable(device, msiCap, 1);
    92→        if (!msiCap && !msixCap) {
    93→            XYLog("%s No MSI cap\n", __FUNCTION__);
    94→            fHalService->release();
    95→            fHalService = NULL;
    96→            return NULL;
    97→        }
    98→        return this;
    99→    }
   100→    return NULL;
   101→}

   103→bool AirportItlwm::configureInterface(IONetworkInterface *netif)
   104→{
   105→    IONetworkData *nd;
   106→    struct _ifnet *ifp = &fHalService->get80211Controller()->ic_ac.ac_if;
   107→    
   108→    if (super::configureInterface(netif) == false) {
   109→        XYLog("super failed\n");
   110→        return false;
   111→    }
   112→    
   113→    nd = netif->getParameter(kIONetworkStatsKey);
   114→    if (!nd || !(fpNetStats = (IONetworkStats *)nd->getBuffer())) {
   115→        XYLog("network statistics buffer unavailable?\n");
   116→        return false;
   117→    }
   118→    ifp->netStat = fpNetStats;
   119→    ether_ifattach(ifp, OSDynamicCast(IOEthernetInterface, netif));
   120→    fpNetStats->collisions = 0;
   121→#ifdef __PRIVATE_SPI__
   122→    netif->configureOutputPullModel(fHalService->getDriverInfo()->getTxQueueSize(), 0, 0, IOEthernetInterface::kOutputPacketSchedulingModelNormal, 0);
   123→#endif
   124→    
   125→    return true;
   126→}

   140→void AirportItlwm::associateSSID(uint8_t *ssid, uint32_t ssid_len, const struct ether_addr &bssid, uint32_t authtype_lower, uint32_t authtype_upper, uint8_t *key, uint32_t key_len, int key_index) 
   141→{
   142→    struct ieee80211com *ic = fHalService->get80211Controller();
   143→    
   144→    ieee80211_disable_rsn(ic);
   145→    ieee80211_disable_wep(ic);
   146→    
   147→    struct ieee80211_wpaparams   wpa;
   148→    struct ieee80211_nwkey               nwkey;
   149→    bzero(&wpa, sizeof(wpa));
   150→    bzero(&nwkey, sizeof(nwkey));
   151→    
   152→    memset(ic->ic_des_essid, 0, IEEE80211_NWID_LEN);
   153→    memcpy(ic->ic_des_essid, ssid, ssid_len);
   154→    ic->ic_des_esslen = ssid_len;
   155→    
   156→    bool is_zero = true;
   157→    for (int i = 0; i < IEEE80211_ADDR_LEN; i++)
   158→    is_zero &= bssid.octet[i] == 0;
   159→    
   160→    if (!is_zero) {
   161→        IEEE80211_ADDR_COPY(ic->ic_des_bssid, bssid.octet);
   162→        ic->ic_flags |= IEEE80211_F_DESBSSID;
   163→    }
   164→    else {
   165→        memset(ic->ic_des_bssid, 0, IEEE80211_ADDR_LEN);        
   166→        ic->ic_flags &= ~IEEE80211_F_DESBSSID;
   167→    }
   168→
   169→    // AUTHTYPE_WPA3_SAE AUTHTYPE_WPA3_FT_SAE
   170→    // we don't really support WPA3, but we have announced we support WPA3 in card capability function. so we fake it as WPA2 to support some WPA2/WPA3 mix wifi connection.
   171→    if (authtype_upper == APPLE80211_AUTHTYPE_WPA3_SAE || authtype_upper == APPLE80211_AUTHTYPE_WPA3_FT_SAE) {
   172→        wpa.i_protos |= IEEE80211_WPA_PROTO_WPA2;
   173→        authtype_upper |= APPLE80211_AUTHTYPE_WPA2_PSK;// hack
   174→    }
   175→    // AUTHTYPE_WPA3_ENTERPRISE AUTHTYPE_WPA3_FT_ENTERPRISE
   176→    if (authtype_upper == APPLE80211_AUTHTYPE_WPA3_ENTERPRISE || authtype_upper == APPLE80211_AUTHTYPE_WPA3_FT_ENTERPRISE) {
   177→        wpa.i_protos |= IEEE80211_WPA_PROTO_WPA2;
   178→        authtype_upper |= APPLE80211_AUTHTYPE_WPA2;// hack
   179→    }
   180→    
   181→    if (authtype_upper & (APPLE80211_AUTHTYPE_WPA | APPLE80211_AUTHTYPE_WPA_PSK | APPLE80211_AUTHTYPE_WPA2 | APPLE80211_AUTHTYPE_WPA2_PSK | APPLE80211_AUTHTYPE_SHA256_PSK | APPLE80211_AUTHTYPE_SHA256_8021X)) {
   182→        XYLog("%s %d\n", __FUNCTION__, __LINE__);
   183→        wpa.i_protos = IEEE80211_WPA_PROTO_WPA1 | IEEE80211_WPA_PROTO_WPA2;
   184→    }
   185→    
   186→    if (authtype_upper & (APPLE80211_AUTHTYPE_WPA_PSK | APPLE80211_AUTHTYPE_WPA2_PSK | APPLE80211_AUTHTYPE_SHA256_PSK)) {
   187→        XYLog("%s %d\n", __FUNCTION__, __LINE__);
   188→        wpa.i_akms |= IEEE80211_WPA_AKM_PSK | IEEE80211_WPA_AKM_SHA256_PSK;
   189→        wpa.i_enabled = 1;
   190→        memcpy(ic->ic_psk, key, sizeof(ic->ic_psk));
   191→        ic->ic_flags |= IEEE80211_F_PSK;
   192→        ieee80211_ioctl_setwpaparms(ic, &wpa);
   193→    }
   194→    if (authtype_upper & (APPLE80211_AUTHTYPE_WPA | APPLE80211_AUTHTYPE_WPA2 | APPLE80211_AUTHTYPE_SHA256_8021X)) {
   195→        XYLog("%s %d\n", __FUNCTION__, __LINE__);
   196→        wpa.i_akms |= IEEE80211_WPA_AKM_8021X | IEEE80211_WPA_AKM_SHA256_8021X; 
   197→        wpa.i_enabled = 1;
   198→        ieee80211_ioctl_setwpaparms(ic, &wpa);
   199→    }
   200→    
   201→    if (authtype_lower == APPLE80211_AUTHTYPE_SHARED) {
   202→        XYLog("shared key authentication is not supported!\n");
   203→        return;
   204→    }
   205→    
   206→    if (authtype_upper == APPLE80211_AUTHTYPE_NONE && authtype_lower == APPLE80211_AUTHTYPE_OPEN) { // Open or WEP Open System
   207→        if (key_len > 0) {
   208→            XYLog("%s %d\n", __FUNCTION__, __LINE__);
   209→            nwkey.i_wepon = IEEE80211_NWKEY_WEP;
   210→            nwkey.i_defkid = key_index + 1;
   211→            nwkey.i_key[key_index].i_keylen = (int)key_len;
   212→            nwkey.i_key[key_index].i_keydat = key;
   213→            ieee80211_ioctl_setnwkeys(ic, &nwkey);
   214→        }
   215→    }
   216→}

   218→void AirportItlwm::setPTK(const u_int8_t *key, size_t key_len) {
   219→    struct ieee80211com *ic = fHalService->get80211Controller();
   220→    struct ieee80211_node       * ni = ic->ic_bss;
   221→    struct ieee80211_key *k;
   222→    int keylen;
   223→    
   224→    ni->ni_rsn_supp_state = RNSA_SUPP_PTKDONE;
   225→    
   226→    if (ni->ni_rsncipher != IEEE80211_CIPHER_USEGROUP) {
   227→        u_int64_t prsc;
   228→        
   229→        /* check that key length matches that of pairwise cipher */
   230→        keylen = ieee80211_cipher_keylen(ni->ni_rsncipher);
   231→        if (key_len != keylen) {
   232→            XYLog("PTK length mismatch. expected %d, got %zu\n", keylen, key_len);
   233→            return;
   234→        }
   235→        prsc = /*(gtk == NULL) ? LE_READ_6(key->rsc) :*/ 0;
   236→        
   237→        /* map PTK to 802.11 key */
   238→        k = &ni->ni_pairwise_key;
   239→        memset(k, 0, sizeof(*k));
   240→        k->k_cipher = ni->ni_rsncipher;
   241→        k->k_rsc[0] = prsc;
   242→        k->k_len = keylen;
   243→        memcpy(k->k_key, key, k->k_len);
   244→        /* install the PTK */
   245→        if ((*ic->ic_set_key)(ic, ni, k) != 0) {
   246→            XYLog("setting PTK failed\n");
   247→            return;
   248→        }
   249→        else
   250→            XYLog("setting PTK successfully\n");
   251→        ni->ni_flags &= ~IEEE80211_NODE_RSN_NEW_PTK;
   252→        ni->ni_flags &= ~IEEE80211_NODE_TXRXPROT;
   253→        ni->ni_flags |= IEEE80211_NODE_RXPROT;
   254→    } else if (ni->ni_rsncipher != IEEE80211_CIPHER_USEGROUP)
   255→        XYLog("%s: unexpected pairwise key update received from %s\n",
   256→              ic->ic_if.if_xname, ether_sprintf(ni->ni_macaddr));
   257→}

   259→void AirportItlwm::setGTK(const u_int8_t *gtk, size_t key_len, u_int8_t kid, u_int8_t *rsc) {
   260→    struct ieee80211com *ic = fHalService->get80211Controller();
   261→    struct ieee80211_node       * ni = ic->ic_bss;
   262→    struct ieee80211_key *k;
   263→    int keylen;
   264→    
   265→    if (gtk != NULL) {
   266→        /* check that key length matches that of group cipher */
   267→        keylen = ieee80211_cipher_keylen(ni->ni_rsngroupcipher);
   268→        if (key_len != keylen) {
   269→            XYLog("GTK length mismatch. expected %d, got %zu\n", keylen, key_len);
   270→            return;
   271→        }
   272→        /* map GTK to 802.11 key */
   273→        k = &ic->ic_nw_keys[kid];
   274→        if (k->k_cipher == IEEE80211_CIPHER_NONE || k->k_len != keylen || memcmp(k->k_key, gtk, keylen) != 0) {
   275→            memset(k, 0, sizeof(*k));
   276→            k->k_id = kid;    /* 0-3 */
   277→            k->k_cipher = ni->ni_rsngroupcipher;
   278→            k->k_flags = IEEE80211_KEY_GROUP;
   279→            //if (gtk[6] & (1 << 2))
   280→            //  k->k_flags |= IEEE80211_KEY_TX;
   281→            k->k_rsc[0] = LE_READ_6(rsc);
   282→            k->k_len = keylen;
   283→            memcpy(k->k_key, gtk, k->k_len);
   284→            /* install the GTK */
   285→            if ((*ic->ic_set_key)(ic, ni, k) != 0) {
   286→                XYLog("setting GTK failed\n");
   287→                return;
   288→            }
   289→            else
   290→                XYLog("setting GTK successfully\n");
   291→        }
   292→    }
   293→    
   294→    if (true) {
   295→        ni->ni_flags |= IEEE80211_NODE_TXRXPROT;
   296→#ifndef IEEE80211_STA_ONLY
   297→        if (ic->ic_opmode != IEEE80211_M_IBSS ||
   298→            ++ni->ni_key_count == 2)
   299→#endif
   300→        {
   301→            XYLog("marking port %s valid\n",
   302→                  ether_sprintf(ni->ni_macaddr));
   303→            ni->ni_port_valid = 1;
   304→            ieee80211_set_link_state(ic, LINK_STATE_UP);
   305→            ni->ni_assoc_fail = 0;
   306→            if (ic->ic_opmode == IEEE80211_M_STA)
   307→                ic->ic_rsngroupcipher = ni->ni_rsngroupcipher;
   308→        }
   309→    }
   310→}

   313→bool AirportItlwm::
   314→createMediumTables(const IONetworkMedium **primary)
   315→{
   316→    IONetworkMedium    *medium;
   317→    
   318→    OSDictionary *mediumDict = OSDictionary::withCapacity(1);
   319→    if (mediumDict == NULL) {
   320→        XYLog("Cannot allocate OSDictionary\n");
   321→        return false;
   322→    }
   323→    
   324→    medium = IONetworkMedium::medium(0x80, 11000000);
   325→    IONetworkMedium::addMedium(mediumDict, medium);
   326→    medium->release();
   327→    if (primary) {
   328→        *primary = medium;
   329→    }
   330→    
   331→    bool result = publishMediumDictionary(mediumDict);
   332→    if (!result)
   333→        XYLog("Cannot publish medium dictionary!\n");
   334→
   335→    mediumDict->release();
   336→    return result;
   337→}

   339→bool AirportItlwm::start(IOService *provider)
   340→{
   341→    int boot_value = 0;
   342→    if (!super::start(provider)) {
   343→        return false;
   344→    }
   345→    if (!serviceMatching("AppleSMC")) {
   346→        super::stop(provider);
   347→        XYLog("No matching AppleSMC dictionary, failing\n");
   348→        return false;
   349→    }
   350→    pciNub = OSDynamicCast(IOPCIDevice, provider);
   351→    if (!pciNub) {
   352→        super::stop(provider);
   353→        return false;
   354→    }
   355→    pciNub->setBusMasterEnable(true);
   356→    pciNub->setIOEnable(true);
   357→    pciNub->setMemoryEnable(true);
   358→    pciNub->configWrite8(0x41, 0);
   359→    if (pciNub->requestPowerDomainState(kIOPMPowerOn,
   360→                                        (IOPowerConnection *) getParentEntry(gIOPowerPlane), IOPMLowestState) != IOPMNoErr) {
   361→        super::stop(provider);
   362→        return false;
   363→    }
   364→    if (initPCIPowerManagment(pciNub) == false) {
   365→        super::stop(pciNub);
   366→        return false;
   367→    }
   368→    if (_fWorkloop == NULL) {
   369→        XYLog("No _fWorkloop!!\n");
   370→        super::stop(pciNub);
   371→        releaseAll();
   372→        return false;
   373→    }
   374→    _fCommandGate = IOCommandGate::commandGate(this, (IOCommandGate::Action)AirportItlwm::tsleepHandler);
   375→    if (_fCommandGate == 0) {
   376→        XYLog("No command gate!!\n");
   377→        super::stop(pciNub);
   378→        releaseAll();
   379→        return false;
   380→    }
   381→    _fWorkloop->addEventSource(_fCommandGate);
   382→    const IONetworkMedium *primaryMedium;
   383→    if (!createMediumTables(&primaryMedium) ||
   384→        !setCurrentMedium(primaryMedium) || !setSelectedMedium(primaryMedium)) {
   385→        XYLog("setup medium fail\n");
   386→        releaseAll();
   387→        return false;
   388→    }
   389→    fHalService->initWithController(this, _fWorkloop, _fCommandGate);
   390→    fHalService->get80211Controller()->ic_event_handler = eventHandler;
   391→    
   392→    if (PE_parse_boot_argn("-novht", &boot_value, sizeof(boot_value)))
   393→        fHalService->get80211Controller()->ic_userflags |= IEEE80211_F_NOVHT;
   394→    if (PE_parse_boot_argn("-noht40", &boot_value, sizeof(boot_value)))
   395→        fHalService->get80211Controller()->ic_userflags |= IEEE80211_F_NOHT40;
   396→    
   397→    if (!fHalService->attach(pciNub)) {
   398→        XYLog("attach fail\n");
   399→        super::stop(pciNub);
   400→        releaseAll();
   401→        return false;
   402→    }
   403→    if (!attachInterface((IONetworkInterface **)&fNetIf, true)) {
   404→        XYLog("attach to interface fail\n");
   405→        fHalService->detach(pciNub);
   406→        super::stop(pciNub);
   407→        releaseAll();
   408→        return false;
   409→    }
   410→    fWatchdogWorkLoop = IOWorkLoop::workLoop();
   411→    if (fWatchdogWorkLoop == NULL) {
   412→        XYLog("init watchdog workloop fail\n");
   413→        fHalService->detach(pciNub);
   414→        super::stop(pciNub);
   415→        releaseAll();
   416→        return false;
   417→    }
   418→    watchdogTimer = IOTimerEventSource::timerEventSource(this, OSMemberFunctionCast(IOTimerEventSource::Action, this, &AirportItlwm::watchdogAction));
   419→    if (!watchdogTimer) {
   420→        XYLog("init watchdog fail\n");
   421→        fHalService->detach(pciNub);
   422→        super::stop(pciNub);
   423→        releaseAll();
   424→        return false;
   425→    }
   426→    fWatchdogWorkLoop->addEventSource(watchdogTimer);
   427→    scanSource = IOTimerEventSource::timerEventSource(this, &fakeScanDone);
   428→    _fWorkloop->addEventSource(scanSource);
   429→    scanSource->enable();
   430→    setLinkStatus(kIONetworkLinkValid);
   431→    if (TAILQ_EMPTY(&fHalService->get80211Controller()->ic_ess))
   432→        fHalService->get80211Controller()->ic_flags |= IEEE80211_F_AUTO_JOIN;
   433→    registerService();
   434→    fNetIf->registerService();
   435→    return true;
   436→}

   461→bool AirportItlwm::initPCIPowerManagment(IOPCIDevice *provider)
   462→{
   463→    UInt16 reg16;
   464→
   465→    reg16 = provider->configRead16(kIOPCIConfigCommand);
   466→
   467→    reg16 |= ( kIOPCICommandBusMaster       |
   468→               kIOPCICommandMemorySpace     |
   469→               kIOPCICommandMemWrInvalidate );
   470→
   471→    reg16 &= ~kIOPCICommandIOSpace;  // disable I/O space
   472→
   473→    provider->configWrite16( kIOPCIConfigCommand, reg16 );
   474→    provider->findPCICapability(kIOPCIPowerManagementCapability,
   475→                                &pmPCICapPtr);
   476→    if (pmPCICapPtr) {
   477→        UInt16 pciPMCReg = provider->configRead32( pmPCICapPtr ) >> 16;
   478→        if (pciPMCReg & kPCIPMCPMESupportFromD3Cold)
   479→            magicPacketSupported = true;
   480→        provider->configWrite16((pmPCICapPtr + 4), 0x8000 );
   481→        IOSleep(10);
   482→    }
   483→    return true;
   484→}

   503→void AirportItlwm::stop(IOService *provider)
   504→{
   505→    XYLog("%s\n", __FUNCTION__);
   506→    struct _ifnet *ifp = &fHalService->get80211Controller()->ic_ac.ac_if;
   507→    super::stop(provider);
   508→    disableAdapter(fNetIf);
   509→    setLinkStatus(kIONetworkLinkValid);
   510→    fHalService->detach(pciNub);
   511→    ether_ifdetach(ifp);
   512→    detachInterface(fNetIf, true);
   513→    OSSafeReleaseNULL(fNetIf);
   514→    releaseAll();

   519→{

   694→UInt32 AirportItlwm::outputPacket(mbuf_t m, void *param)
   695→{
   696→//    XYLog("%s\n", __FUNCTION__);
   697→    IOReturn ret = kIOReturnOutputSuccess;
   698→    struct _ifnet *ifp = &fHalService->get80211Controller()->ic_ac.ac_if;
   699→    
   700→    if (fHalService->get80211Controller()->ic_state != IEEE80211_S_RUN || ifp->if_snd.queue == NULL) {
   701→        if (m && mbuf_type(m) != MBUF_TYPE_FREE)
   702→            freePacket(m);
   703→        return kIOReturnOutputDropped;
   704→    }
   705→    if (m == NULL) {
   706→        XYLog("%s m==NULL!!\n", __FUNCTION__);
   707→        ifp->netStat->outputErrors++;
   708→        ret = kIOReturnOutputDropped;
   709→    }
   710→    if (!(mbuf_flags(m) & MBUF_PKTHDR) ){
   711→        XYLog("%s pkthdr is NULL!!\n", __FUNCTION__);
   712→        ifp->netStat->outputErrors++;
   713→        freePacket(m);
   714→        ret = kIOReturnOutputDropped;
   715→    }
   716→    if (mbuf_type(m) == MBUF_TYPE_FREE) {
   717→        XYLog("%s mbuf is FREE!!\n", __FUNCTION__);
   718→        ifp->netStat->outputErrors++;
   719→        ret = kIOReturnOutputDropped;
   720→    }
   721→    if (!ifp->if_snd.queue->lockEnqueue(m)) {
   722→        freePacket(m);
   723→        ret = kIOReturnOutputDropped;
   724→    }
   725→    (*ifp->if_start)(ifp);
   726→    return ret;
   727→}
